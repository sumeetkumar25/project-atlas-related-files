# atlas/core/console.py
"""
Atlas Console Utilities

Centralized console encoding configuration and Unicode-safe symbols.
Import setup_console() in any module that prints to console.
"""

import sys

# Unicode-safe status symbols
USE_UNICODE = True

SUCCESS = "✓" if USE_UNICODE else "[OK]"
FAIL = "✗" if USE_UNICODE else "[FAIL]"
WARNING = "⚠" if USE_UNICODE else "[WARN]"
ARROW = "→" if USE_UNICODE else "->"
SECTION = "─" if USE_UNICODE else "-"
BLOCK = "█" if USE_UNICODE else "#"


def setup_console():
    """
    Configure console encoding for cross-platform compatibility.
    
    Call this once at the top of any executable entry point
    or shared module that prints to console.
    
    Safe to call multiple times - subsequent calls are no-ops.
    """
    if hasattr(setup_console, "_called"):
        return
    
    try:
        sys.stdout.reconfigure(encoding="utf-8")
        sys.stderr.reconfigure(encoding="utf-8")
    except Exception:
        pass
    
    setup_console._called = True
