"""
Rolling market features: causal rolling Hurst exponent and normalized
Shannon entropy.

Both features are computed at daily resolution:

    For each calendar day D with enough trailing history,
    a statistic is computed on the log returns observed in the
    window [D - window_days, D). The resulting value is broadcast
    to every intraday bar of day D.

This makes the output causal: every value at row t depends only on
observations with timestamp < the calendar day of t.

Why per-day instead of per-bar:
  - A 60-day rolling Hurst / entropy changes negligibly within a day.
  - Per-bar computation on 50,000+ rows of R/S analysis would be
    prohibitively slow for zero additional information.
  - Per-day broadcast is the standard treatment for slow-moving
    regime features in quantitative research.
"""

import numpy as np
import pandas as pd


WINDOW_DAYS = 60


def _prepare_rolling_inputs(timestamps, close):
    """
    Convert inputs to aligned Series, compute gap-aware log returns,
    and derive calendar-day boundaries.
    """
    ts = pd.to_datetime(
        pd.Series(timestamps).reset_index(drop=True),
        errors="coerce",
    )
    px = pd.to_numeric(
        pd.Series(close).reset_index(drop=True),
        errors="coerce",
    )

    n = len(px)

    if n == 0:
        return ts, px, np.array([], dtype=float), pd.Series(dtype="datetime64[ns]")

    px_arr = px.to_numpy(dtype=float)
    ts_arr = ts.to_numpy(dtype="datetime64[ns]")

    log_ret = np.full(n, np.nan, dtype=float)

    if n >= 2:
        expected_interval = pd.Timedelta(
            seconds=900
        )  # BTCUSD 15m default

        # Infer timeframe from the dominant positive timestamp difference.
        differences = pd.Series(ts).diff()
        valid_differences = differences[
            differences > pd.Timedelta(0)
        ]

        if not valid_differences.empty:
            inferred = valid_differences.mode()
            if not inferred.empty:
                expected_interval = inferred.iloc[0]

        # Gap-aware adjacent return computation.
        #
        # A return spanning a timestamp gap larger than the expected
        # interval is not a single-bar return — it is a multi-bar
        # aggregate move compressed into one observation. Treating it
        # as a unit-interval return would bias the log-returns
        # distribution that Hurst (R/S) and entropy both analyze under
        # the implicit assumption of uniform sampling.
        #
        # We compute every adjacent ratio, then NaN out the ones whose
        # timestamp gap exceeds 1.5 * expected_interval.
        with np.errstate(divide="ignore", invalid="ignore"):
            ratios = px_arr[1:] / px_arr[:-1]

        candidate_log_ret = np.log(ratios)

        # Per-adjacent-pair gap (length n-1). diff()[0] is NaT for the
        # first element; slice from index 1 so the arrays align with
        # candidate_log_ret, which represents pairs (0,1), (1,2), ...
        gap_series = pd.Series(ts_arr).diff().iloc[1:]

        gap_tolerance = expected_interval * 1.5

        is_contiguous = (
            gap_series.notna()
            & (gap_series <= gap_tolerance)
        ).to_numpy()

        log_ret[1:] = np.where(
            is_contiguous,
            candidate_log_ret,
            np.nan,
        )

    day_index = ts.dt.floor("D")

    return ts, px, log_ret, day_index


def _rolling_statistic(
    timestamps,
    close,
    window_days,
    statistic_fn,
):
    """
    Shared causal rolling-window computation.

    For each calendar day with sufficient trailing history, calls
    statistic_fn(window_returns) on the log returns observed in
    [D - window_days, D), then assigns that value to every bar of D.

    statistic_fn must return a float; NaN or exceptions skip the day.
    """
    ts, px, log_ret, day_index = _prepare_rolling_inputs(timestamps, close)

    # ----------------------------------------------------------
    # Timestamp integrity checks.
    #
    # A rolling-window computation is meaningless if the input
    # timestamps are NaT, unordered, or contain duplicates:
    #   - NaT rows corrupt every downstream comparison.
    #   - Unordered input violates the "past-only" contract of
    #     the rolling window — the window would silently contain
    #     future rows.
    #   - Duplicate timestamps double-count observations inside
    #     every window, biasing Hurst and entropy estimates.
    #
    # Failing fast here surfaces the defect at the source rather
    # than producing biased rolling features silently.
    # ----------------------------------------------------------
    if ts.isna().any():
        raise ValueError("Rolling features cannot use invalid timestamps.")

    if not ts.is_monotonic_increasing:
        raise ValueError("Rolling feature timestamps must be chronological.")

    if ts.duplicated().any():
        raise ValueError("Duplicate timestamps found in rolling feature input.")

    first_timestamp = ts.iloc[0]

    n = len(px)
    out = np.full(n, np.nan, dtype=float)

    if n == 0:
        return out

    ts_array = ts.to_numpy()
    day_array = day_index.to_numpy()

    unique_days = pd.Series(day_array).dropna().unique()

    for day in unique_days:
        day_start = pd.Timestamp(day)
        window_start = day_start - pd.Timedelta(days=window_days)

        # Require a complete window_days-calendar-day history.
        #
        # If the requested window extends before the first
        # observation, the statistic would be computed on a
        # partial window — and its value would be emitted to
        # downstream consumers as if it were a full-window
        # estimate. Skipping here keeps every non-NaN output
        # row backed by a complete window of history.
        if window_start < first_timestamp:
            continue

        in_window = (ts_array >= np.datetime64(window_start)) & (
            ts_array < np.datetime64(day_start)
        )

        window_returns = log_ret[in_window]
        window_returns = window_returns[np.isfinite(window_returns)]

        # ------------------------------------------------------
        # Require at least 99% of expected return observations.
        #
        # Missing candles must not silently become valid history.
        # A 60-day window with 50 missing bars is not the same
        # sample as a 60-day window with none, but without this
        # gate both would emit "a 60-day rolling Hurst value."
        # ------------------------------------------------------------------
        expected_interval = pd.Series(ts).diff()
        expected_interval = expected_interval[
            expected_interval > pd.Timedelta(0)
        ].mode()

        if expected_interval.empty:
            continue

        interval = expected_interval.iloc[0]
        expected_returns = int(
            pd.Timedelta(days=window_days) / interval
        )
        minimum_returns = int(np.ceil(expected_returns * 0.99))

        if window_returns.size < minimum_returns:
            continue

        # Need at least a few observations to estimate anything.
        # The underlying engine will raise on truly-insufficient data;
        # this guard skips the obvious warm-up case cheaply.
        if window_returns.size < 10:
            continue

        try:
            value = statistic_fn(window_returns)
        except Exception:
            continue

        try:
            value = float(value)
        except (TypeError, ValueError):
            continue

        if np.isfinite(value):
            bars_of_day = day_array == day
            out[bars_of_day] = value

    return out


def compute_rolling_hurst(
    timestamps,
    close,
    window_days=WINDOW_DAYS,
):
    """
    Compute a causal rolling Hurst exponent.

    Returns a float64 ndarray of the same length as `close`. Rows
    before the first day with `window_days` of trailing history
    are NaN.

    Raises no exception on insufficient data — those rows are NaN.
    """
    # Lazy import avoids importing Module 035 while its package
    # is still initializing.
    from atlas.modules.module035_hurst.engine import HurstEngine

    engine = HurstEngine(
        min_window=16,
        num_windows=20,
    )

    def _score(window_returns):
        result = engine.compute(
            window_returns,
            series_type="log_returns",
        )
        return result.hurst_exponent

    return _rolling_statistic(
        timestamps,
        close,
        window_days,
        _score,
    )


def compute_rolling_entropy(
    timestamps,
    close,
    window_days=WINDOW_DAYS,
):
    """
    Compute a causal rolling normalized Shannon entropy.

    Returns a float64 ndarray of the same length as `close`. Values
    are in [0, 1]; rows before the first day with `window_days` of
    trailing history are NaN.
    """
    # Lazy import avoids importing Module 037 while its package
    # is still initializing. Same circular-import class as Module 035.
    from atlas.modules.module037_entropy.engine import EntropyEngine

    engine = EntropyEngine(
        bins="fd",
    )

    def _score(window_returns):
        result = engine.compute(
            window_returns,
            series_type="log_returns",
        )
        return result.normalized_entropy

    return _rolling_statistic(
        timestamps,
        close,
        window_days,
        _score,
    )
