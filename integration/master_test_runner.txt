"""
Master Test Runner - Project Atlas

Automatically discovers and executes all module test files sequentially.

Features:
- Automatic module discovery (no hardcoded list)
- UTF-8 encoding support
- Clean PASS/FAIL output (no Unicode issues)
- Memory and runtime tracking
- Detailed report generation
- Re-run failed tests
- Diagnostic timing to identify hangs
- Smart module matching for single module runs
- Configurable timeout via command line

Usage:
    python -m atlas.integration.master_test_runner
    python -m atlas.integration.master_test_runner --single module111
    python -m atlas.integration.master_test_runner --timeout 900
    python -m atlas.integration.master_test_runner --timeout 1800 --no-re-run
"""

import subprocess
import sys
import time
import psutil
from pathlib import Path
from datetime import datetime
import os
import argparse
import warnings
warnings.filterwarnings('ignore')

# Add project root to path
PROJECT_ROOT = Path(__file__).parent.parent.parent
sys.path.insert(0, str(PROJECT_ROOT))

# Default timeout for research framework
DEFAULT_TIMEOUT = 900  # 15 minutes per module


class MasterTestRunner:
    """Master test runner for Project Atlas with automatic discovery."""
    
    def __init__(self, timeout_seconds: int = None):
        """
        Initialize the master test runner.
        
        Args:
            timeout_seconds: Timeout per module in seconds (default: 900)
        """
        if timeout_seconds is None:
            timeout_seconds = DEFAULT_TIMEOUT
            
        self.process = psutil.Process()
        self.peak_memory = 0
        self.start_time = None
        self.end_time = None
        self.results = {}
        self.failed_modules = []
        self.passed_modules = []
        self.skipped_modules = []
        self.timeout_seconds = timeout_seconds
        
        # Auto-discover all modules with test_module.py
        self.tests = self.discover_modules()
        
        print(f"[DIAGNOSTIC] Master runner initialized with timeout={self.timeout_seconds}s")
        print(f"[DIAGNOSTIC] Python executable: {sys.executable}")
        print(f"[DIAGNOSTIC] Project root: {PROJECT_ROOT}")
    
    def discover_modules(self) -> list:
        """
        Automatically discover all modules with test_module.py.
        
        Returns:
            list: Sorted list of module test paths
        """
        modules = []
        module_root = PROJECT_ROOT / "atlas" / "modules"
        
        if not module_root.exists():
            print(f"⚠ Module root not found: {module_root}")
            return []
        
        for folder in sorted(module_root.iterdir()):
            if folder.is_dir():
                # Check if it's a module directory (starts with 'module')
                if folder.name.startswith('module'):
                    test_file = folder / "test_module.py"
                    if test_file.exists():
                        modules.append(f"atlas.modules.{folder.name}.test_module")
        
        print(f"[DIAGNOSTIC] Discovered {len(modules)} modules with test_module.py")
        return modules
    
    def track_memory(self):
        """Track peak memory usage."""
        try:
            memory = self.process.memory_info().rss / (1024 * 1024)  # MB
            if memory > self.peak_memory:
                self.peak_memory = memory
        except:
            pass
    
    def run_test(self, test_name: str) -> tuple:
        """
        Run a single test module with diagnostic timing.
        
        Args:
            test_name: Module test path (e.g., "atlas.modules.module001.test_module")
        
        Returns:
            tuple: (success, output, error, duration)
        """
        print(f"\n[DIAGNOSTIC] Starting: {test_name}")
        print(f"[DIAGNOSTIC] Timeout: {self.timeout_seconds}s")
        
        t0 = time.perf_counter()
        
        try:
            # Run the test using subprocess with proper encoding
            result = subprocess.run(
                [sys.executable, "-m", test_name],
                capture_output=True,
                text=True,
                encoding="utf-8",
                errors="replace",
                timeout=self.timeout_seconds,
                env=os.environ.copy()
            )
            
            t1 = time.perf_counter()
            duration = t1 - t0
            
            print(f"[DIAGNOSTIC] Finished subprocess after {duration:.2f} sec")
            print(f"[DIAGNOSTIC] Return code: {result.returncode}")
            print(f"[DIAGNOSTIC] stdout length: {len(result.stdout)} chars")
            print(f"[DIAGNOSTIC] stderr length: {len(result.stderr)} chars")
            
            success = result.returncode == 0
            
            # Track memory
            self.track_memory()
            
            return success, result.stdout, result.stderr, duration
            
        except subprocess.TimeoutExpired as e:
            t1 = time.perf_counter()
            duration = t1 - t0
            
            print(f"[DIAGNOSTIC] TIMEOUT after {duration:.2f} sec (limit was {self.timeout_seconds}s)")
            print(f"[DIAGNOSTIC] Partial stdout length: {len(e.stdout or '')} chars")
            print(f"[DIAGNOSTIC] Partial stderr length: {len(e.stderr or '')} chars")
            
            # Try to get process info if available
            try:
                import psutil as ps
                for proc in ps.process_iter(['pid', 'name', 'cmdline']):
                    try:
                        if proc.info['cmdline'] and any(test_name in str(cmd) for cmd in proc.info['cmdline']):
                            print(f"[DIAGNOSTIC] Found hanging process: PID={proc.info['pid']}, Name={proc.info['name']}")
                            print(f"[DIAGNOSTIC] Cmdline: {proc.info['cmdline']}")
                    except:
                        pass
            except:
                pass
            
            return False, e.stdout or "", e.stderr or f"Timeout after {duration:.2f}s", duration
            
        except Exception as e:
            t1 = time.perf_counter()
            duration = t1 - t0
            
            print(f"[DIAGNOSTIC] EXCEPTION after {duration:.2f} sec: {type(e).__name__}: {str(e)}")
            
            return False, "", str(e), duration
    
    def run_single_module(self, module_name: str):
        """
        Run a single module for diagnostic purposes.
        Smart matching: finds the discovered module that matches the given name.
        
        Args:
            module_name: Module name to match (e.g., "module111")
        
        Returns:
            bool: True if test passed, False otherwise
        """
        # Find matching discovered modules
        matches = [
            t for t in self.tests
            if t.split(".")[-2].startswith(module_name)
        ]
        
        if not matches:
            print(f"\n[ERROR] No discovered module matching '{module_name}'")
            print("  Available modules (first 10):")
            for m in self.tests[:10]:
                print(f"    {m.split('.')[-2]}")
            if len(self.tests) > 10:
                print(f"    ... and {len(self.tests) - 10} more")
            return False
        
        if len(matches) > 1:
            print(f"\n[ERROR] Multiple matches for '{module_name}':")
            for m in matches:
                print(f"    {m}")
            print("\n  Please be more specific:")
            for m in matches:
                print(f"    python -m atlas.integration.master_test_runner --single {m.split('.')[-2]}")
            return False
        
        test_name = matches[0]
        module_display = test_name.split('.')[-2]
        
        print(f"\n[DIAGNOSTIC] Running single module: {module_display}")
        print(f"[DIAGNOSTIC] Test path: {test_name}")
        print(f"[DIAGNOSTIC] Timeout: {self.timeout_seconds}s")
        print("-" * 60)
        
        success, stdout, stderr, duration = self.run_test(test_name)
        
        print("\n" + "=" * 60)
        print(f"RESULT: {module_display}")
        print("=" * 60)
        print(f"  Status: {'PASS' if success else 'FAIL'}")
        print(f"  Duration: {duration:.2f}s")
        print(f"  Stdout length: {len(stdout)} chars")
        print(f"  Stderr length: {len(stderr)} chars")
        
        if stdout:
            print("\n  STDOUT (first 1000 chars):")
            print("-" * 40)
            print(stdout[:1000])
        if stderr:
            print("\n  STDERR (first 500 chars):")
            print("-" * 40)
            print(stderr[:500])
        
        return success
    
    def run_all(self):
        """Run all discovered module tests."""
        total_tests = len(self.tests)
        
        print("=" * 60)
        print("PROJECT ATLAS MASTER VALIDATION")
        print("=" * 60)
        print(f"\nStart Time: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        print(f"Modules Discovered: {total_tests}")
        print(f"Module Timeout: {self.timeout_seconds}s")
        print("\n" + "-" * 60)
        
        self.start_time = time.time()
        
        for i, test_name in enumerate(self.tests, 1):
            # Extract module name for display
            module_parts = test_name.split('.')
            module_name = module_parts[-2] if len(module_parts) >= 2 else test_name
            
            print(f"\n[{i:03d}/{total_tests}] Running {module_name}...")
            sys.stdout.flush()
            
            success, stdout, stderr, duration = self.run_test(test_name)
            
            self.results[test_name] = {
                'success': success,
                'duration': duration,
                'stdout': stdout,
                'stderr': stderr
            }
            
            if success:
                print(f"[{i:03d}] {module_name}: PASS ({duration:.2f}s)")
                self.passed_modules.append(test_name)
            else:
                print(f"[{i:03d}] {module_name}: FAIL ({duration:.2f}s)")
                self.failed_modules.append(test_name)
                
                # Print error details
                if stderr:
                    print(f"\n    Error: {stderr[:500]}...")
                if stdout and "FAIL" in stdout:
                    print(f"\n    Output: {stdout[:500]}...")
        
        self.end_time = time.time()
        
        # Print summary
        self.print_summary()
        
        # Save report
        self.save_report()
        
        return len(self.failed_modules) == 0
    
    def print_summary(self):
        """Print test summary with PASS/FAIL (no Unicode)."""
        total = len(self.tests)
        passed = len(self.passed_modules)
        failed = len(self.failed_modules)
        skipped = len(self.skipped_modules)
        total_time = self.end_time - self.start_time
        
        print("\n" + "=" * 60)
        print("VALIDATION SUMMARY")
        print("=" * 60)
        print(f"\n  Modules Discovered: {total}")
        print(f"  Passed: {passed}")
        print(f"  Failed: {failed}")
        print(f"  Skipped: {skipped}")
        print(f"  Runtime: {total_time:.1f} seconds ({total_time/60:.1f} minutes)")
        print(f"  Peak Memory: {self.peak_memory:.0f} MB")
        print(f"  Module Timeout: {self.timeout_seconds}s")
        
        if self.failed_modules:
            print("\n  FAILED MODULES:")
            for test in self.failed_modules:
                module_name = test.split('.')[-2]
                duration = self.results[test]['duration']
                print(f"    [FAIL] {module_name} ({duration:.2f}s)")
        
        status = "PASSED" if failed == 0 else "FAILED"
        print(f"\n  Status: {status}")
        print("\n" + "=" * 60)
    
    def save_report(self):
        """Save validation report to file."""
        report_dir = Path("reports")
        report_dir.mkdir(exist_ok=True)
        
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        report_path = report_dir / f"master_validation_{timestamp}.txt"
        
        total = len(self.tests)
        passed = len(self.passed_modules)
        failed = len(self.failed_modules)
        total_time = self.end_time - self.start_time
        
        with open(report_path, "w", encoding="utf-8") as f:
            f.write("=" * 60 + "\n")
            f.write("PROJECT ATLAS MASTER VALIDATION REPORT\n")
            f.write("=" * 60 + "\n\n")
            
            f.write(f"Date: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
            f.write(f"Modules Discovered: {total}\n")
            f.write(f"Passed: {passed}\n")
            f.write(f"Failed: {failed}\n")
            f.write(f"Runtime: {total_time:.1f} seconds ({total_time/60:.1f} minutes)\n")
            f.write(f"Peak Memory: {self.peak_memory:.0f} MB\n")
            f.write(f"Module Timeout: {self.timeout_seconds}s\n\n")
            
            f.write("-" * 60 + "\n")
            f.write("MODULE RESULTS\n")
            f.write("-" * 60 + "\n\n")
            
            for test_name, result in self.results.items():
                module_name = test_name.split('.')[-2]
                status = "PASS" if result['success'] else "FAIL"
                f.write(f"{module_name:<40} {status:<10} {result['duration']:>8.2f}s\n")
            
            if self.failed_modules:
                f.write("\n" + "-" * 60 + "\n")
                f.write("FAILED MODULES\n")
                f.write("-" * 60 + "\n\n")
                for test in self.failed_modules:
                    module_name = test.split('.')[-2]
                    duration = self.results[test]['duration']
                    f.write(f"  [FAIL] {module_name} ({duration:.2f}s)\n")
                    if self.results[test]['stderr']:
                        f.write(f"    Error: {self.results[test]['stderr'][:500]}\n")
            
            status = "PASSED" if failed == 0 else "FAILED"
            f.write("\n" + "=" * 60 + "\n")
            f.write(f"OVERALL STATUS: {status}\n")
            f.write("=" * 60 + "\n")
        
        print(f"\n  Report saved to: {report_path}")
    
    def run_failed_tests(self):
        """Re-run only failed tests."""
        if not self.failed_modules:
            print("\n  No failed tests to re-run.")
            return True
        
        print("\n" + "=" * 60)
        print("RE-RUNNING FAILED TESTS")
        print("=" * 60)
        
        re_run_results = {}
        re_run_failed = []
        
        for test_name in self.failed_modules:
            module_name = test_name.split('.')[-2]
            print(f"\n  Re-running {module_name}...")
            sys.stdout.flush()
            
            success, stdout, stderr, duration = self.run_test(test_name)
            
            if success:
                print(f"  {module_name}: PASS ({duration:.2f}s)")
                re_run_results[test_name] = True
            else:
                print(f"  {module_name}: FAIL ({duration:.2f}s)")
                re_run_results[test_name] = False
                re_run_failed.append(test_name)
        
        # Update results
        for test_name, success in re_run_results.items():
            if success and test_name in self.failed_modules:
                self.failed_modules.remove(test_name)
                self.passed_modules.append(test_name)
                self.results[test_name]['success'] = True
        
        print(f"\n  Re-run: {len(re_run_results)} tests, {len(re_run_failed)} still failed")
        
        return len(re_run_failed) == 0


def main():
    """Main entry point with configurable timeout."""
    
    parser = argparse.ArgumentParser(
        description="Master Test Runner for Project Atlas",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
Examples:
  python -m atlas.integration.master_test_runner
  python -m atlas.integration.master_test_runner --timeout 600
  python -m atlas.integration.master_test_runner --single module111
  python -m atlas.integration.master_test_runner --timeout 1800 --no-re-run
        """
    )
    parser.add_argument(
        "--timeout",
        type=int,
        default=None,
        help=f"Timeout per module in seconds (default: {DEFAULT_TIMEOUT})"
    )
    parser.add_argument(
        "--single",
        type=str,
        help="Run a single module (e.g., module111) for diagnostic purposes"
    )
    parser.add_argument(
        "--no-re-run",
        action="store_true",
        help="Don't ask to re-run failed tests"
    )
    
    args = parser.parse_args()
    
    # Use provided timeout or default
    timeout = args.timeout if args.timeout is not None else DEFAULT_TIMEOUT
    
    print(f"[DIAGNOSTIC] Using timeout: {timeout}s")
    
    try:
        runner = MasterTestRunner(timeout_seconds=timeout)
        
        if not runner.tests:
            print("  No test modules found!")
            print(f"  Checked: {PROJECT_ROOT / 'atlas' / 'modules'}")
            sys.exit(1)
        
        # Run single module if specified
        if args.single:
            success = runner.run_single_module(args.single)
            sys.exit(0 if success else 1)
        
        # Run all modules
        success = runner.run_all()
        
        # If there are failures and not suppressed, ask if user wants to re-run them
        if runner.failed_modules and not args.no_re_run:
            print("\n  Some tests failed. Do you want to re-run them? (y/n): ", end="")
            response = input().strip().lower()
            if response == 'y':
                runner.run_failed_tests()
        
        if success or len(runner.failed_modules) == 0:
            print("\n[PASS] MASTER VALIDATION PASSED")
            sys.exit(0)
        else:
            print("\n[FAIL] MASTER VALIDATION FAILED")
            print(f"   Failed modules: {len(runner.failed_modules)}")
            sys.exit(1)
            
    except KeyboardInterrupt:
        print("\n\n  Validation interrupted by user.")
        sys.exit(1)
    except Exception as e:
        print(f"\n  Fatal error: {e}")
        import traceback
        traceback.print_exc()
        sys.exit(1)


if __name__ == "__main__":
    main()
