"""
Dependency Injector - Builds feature DataFrames from dependencies
with dataset awareness.
"""

from typing import List, Optional, Dict
import pandas as pd

from .artifact_store import ArtifactStore


class DependencyInjector:
    """
    Builds combined feature DataFrames from dependent modules.

    Every module must export 'features' in its artifacts.
    Dataset-aware: only uses artifacts from the same dataset.
    """

    def __init__(self, artifact_store: ArtifactStore):
        self.artifact_store = artifact_store
        self._injection_log = []

    def build_feature_dataframe(
        self,
        dataset: str,
        dependencies: List[str],
    ) -> Optional[pd.DataFrame]:

        if not dependencies:
            return None

        feature_frames = []
        injection_details = []

        for dep in dependencies:
            artifacts = self.artifact_store.get(dataset, dep)

            if not artifacts:
                raise RuntimeError(
                    f"No artifacts found for dependency "
                    f"{dep} on dataset {dataset}"
                )

            if "features" not in artifacts:
                raise RuntimeError(
                    f"Module {dep} did not export 'features' "
                    f"on dataset {dataset}. "
                    f"Available: {list(artifacts.keys())}"
                )

            feature_df = artifacts["features"]

            if not isinstance(feature_df, pd.DataFrame):
                raise TypeError(
                    f"Features from module {dep} must be a DataFrame"
                )

            feature_frames.append(feature_df)

            injection_details.append({
                "dependency": dep,
                "columns": list(feature_df.columns),
                "rows": len(feature_df),
                "artifact_keys": list(artifacts.keys()),
            })

        if not feature_frames:
            return None

        self._injection_log.append({
            "dataset": dataset,
            "dependencies": injection_details,
            "total_features": sum(
                len(frame.columns) for frame in feature_frames
            ),
        })

        # ---------------------------------------------------------
        # Reference frame — column uniqueness and timestamp extraction
        # ---------------------------------------------------------
        reference = feature_frames[0]
        reference_dep = dependencies[0]

        if not reference.columns.is_unique:
            raise ValueError(
                f"Dependency {reference_dep} contains duplicate "
                f"column names for dataset {dataset}"
            )

        if "timestamp" in reference.columns:
            reference_ts = pd.to_datetime(
                reference["timestamp"],
                errors="coerce",
            ).reset_index(drop=True)

            if reference_ts.isna().any():
                raise ValueError(
                    f"Invalid timestamps in dependency {reference_dep} "
                    f"for dataset {dataset}"
                )
        else:
            reference_ts = None

        # ---------------------------------------------------------
        # Validate every dependency before assembling anything
        # ---------------------------------------------------------
        for dep_id, frame in zip(dependencies[1:], feature_frames[1:]):
            if not frame.columns.is_unique:
                raise ValueError(
                    f"Dependency {dep_id} contains duplicate "
                    f"column names for dataset {dataset}"
                )

            if len(frame) != len(reference):
                raise ValueError(
                    f"DEPENDENCY ROW COUNT MISMATCH: "
                    f"{dataset}: {reference_dep}={len(reference)}, "
                    f"{dep_id}={len(frame)}"
                )

            if (
                reference_ts is not None
                and "timestamp" in frame.columns
            ):
                current_ts = pd.to_datetime(
                    frame["timestamp"],
                    errors="coerce",
                ).reset_index(drop=True)

                if (
                    current_ts.isna().any()
                    or not reference_ts.equals(current_ts)
                ):
                    raise ValueError(
                        f"DEPENDENCY TIMESTAMP MISMATCH: "
                        f"{dataset}: {reference_dep} vs {dep_id}"
                    )

        # ---------------------------------------------------------
        # Assemble the combined DataFrame.
        #
        # 1. Reset indexes so concat aligns positionally (not on
        #    index labels — which can silently union and pad NaN
        #    if any frame carries a non-default index).
        # 2. Retain exactly one timestamp column (the reference's).
        #    Every other dependency's timestamp is dropped.
        # 3. Preserve all other feature columns unchanged.
        # ---------------------------------------------------------
        aligned_frames = [
            reference.reset_index(drop=True)
        ]

        for frame in feature_frames[1:]:
            frame_copy = frame.reset_index(drop=True).copy()

            if "timestamp" in frame_copy.columns:
                frame_copy = frame_copy.drop(columns=["timestamp"])

            aligned_frames.append(frame_copy)

        combined = pd.concat(aligned_frames, axis=1)

        # ---------------------------------------------------------
        # Duplicate-column handling.
        #
        # Columns that collide across dependencies are prefixed
        # with their source dependency ID.
        # ---------------------------------------------------------
        duplicates = combined.columns.duplicated()

        if duplicates.any():
            duplicate_cols = combined.columns[duplicates].tolist()

            for col in duplicate_cols:
                for i, frame in enumerate(aligned_frames):
                    if col in frame.columns:
                        dep_id = dependencies[i]
                        combined.rename(
                            columns={col: f"{dep_id}_{col}"},
                            inplace=True,
                        )
                        break

        return combined

    def validate_dependencies(
        self,
        dataset: str,
        module_id: str,
        dependencies: List[str],
        completed_modules: set,
    ) -> bool:

        for dep_id in dependencies:
            dep_key = (dataset, dep_id)

            if dep_key not in completed_modules:
                raise RuntimeError(
                    f"Module {module_id} on {dataset} requires "
                    f"{dep_id}, but it has not been completed"
                )

            if not self.artifact_store.has_module(dataset, dep_id):
                raise RuntimeError(
                    f"Module {module_id} on {dataset} requires "
                    f"{dep_id}, but no artifacts found"
                )

            if not self.artifact_store.has(
                dataset, dep_id, "features"
            ):
                raise RuntimeError(
                    f"Module {module_id} on {dataset} requires "
                    f"{dep_id}, but it did not export 'features'"
                )

            features = self.artifact_store.get_features(
                dataset, dep_id
            )

            if features is not None and features.empty:
                raise RuntimeError(
                    f"Module {module_id} on {dataset} requires "
                    f"{dep_id}, but its features DataFrame is empty"
                )

        return True

    def get_injection_log(self) -> List[Dict]:
        return self._injection_log.copy()

    def clear_log(self) -> None:
        self._injection_log.clear()

    def log_dependency_summary(
        self,
        dataset: str,
        module_id: str,
        dependencies: List[str],
    ) -> None:

        print(f"\n{'=' * 60}")
        print("DEPENDENCY INJECTION")
        print(f"{'=' * 60}")
        print(f"Module: {module_id}")
        print(f"Dataset: {dataset}")
        print(f"Dependencies: {dependencies}")
        print("-" * 40)

        for dep in dependencies:
            artifacts = self.artifact_store.get(dataset, dep)

            if artifacts and "features" in artifacts:
                df = artifacts["features"]

                print(
                    f"✓ {dep}: {len(df.columns)} columns, "
                    f"{len(df)} rows"
                )
                print(
                    f"  Columns: {list(df.columns)[:5]}"
                    f"{'...' if len(df.columns) > 5 else ''}"
                )
                print(
                    f"  Has validation: {'validation' in artifacts}"
                )
                print(
                    f"  Has statistics: {'statistics' in artifacts}"
                )
            else:
                print(f"✗ {dep}: NOT FOUND or missing 'features'")

        print("=" * 60)

    def get_feature_columns(
        self,
        dataset: str,
        dependencies: List[str],
    ) -> Dict[str, List[str]]:

        result = {}

        for dep in dependencies:
            features = self.artifact_store.get_features(
                dataset, dep
            )

            if features is not None:
                result[dep] = list(features.columns)

        return result
