"""
Research Runner - Main entry point for research campaigns.
Dataset-aware execution with dependency injection.
"""

import sys
import time
from pathlib import Path
from datetime import datetime
import pandas as pd
import traceback
import json

# Add project root to path
PROJECT_ROOT = Path(__file__).resolve().parents[2]
DEFAULT_DATA_PATH = PROJECT_ROOT / "data" / "raw"

sys.path.insert(0, str(PROJECT_ROOT))

from atlas.research.dataset_registry import DatasetRegistry
from atlas.research.module_loader import ModuleLoader
from atlas.research.export_manager import ExportManager
from atlas.research.research_logger import ResearchLogger
from atlas.research.summary_builder import SummaryBuilder
from atlas.core.result_utils import result_succeeded

from atlas.framework.artifact_store import ArtifactStore
from atlas.framework.dependency_injector import DependencyInjector

from atlas.research.target_manager import TargetManager

from atlas.research.topological_sort import (
    topological_sort,
    get_execution_order,
)


class ResearchRunner:
    """Main runner for research campaigns with dataset-aware dependency injection."""
    
    def __init__(
        self,
        data_path: Path = None,
        output_dir: Path = None,
        log_dir: Path = None,
        modules_filter: list = None
    ):
        """
        Initialize the research runner.
        
        Args:
            data_path: Path to raw data directory
            output_dir: Path to output directory
            log_dir: Path to log directory
            modules_filter: List of module IDs to run (None = all)
        """
        if data_path is None:
            data_path = DEFAULT_DATA_PATH
        if output_dir is None:
            output_dir = PROJECT_ROOT / "research_results"
        if log_dir is None:
            log_dir = output_dir / "logs"
        
        self.data_path = Path(data_path)
        self.output_dir = Path(output_dir)
        self.log_dir = Path(log_dir)
        self.modules_filter = modules_filter
        
        # Initialize components
        self.dataset_registry = DatasetRegistry(self.data_path)
        self.module_loader = ModuleLoader()
        self.export_manager = ExportManager(self.output_dir)
        self.logger = ResearchLogger(self.log_dir)
        self.summary_builder = SummaryBuilder(self.output_dir)
        
        # Artifact store and dependency injector
        self.artifact_store = ArtifactStore()
        self.dependency_injector = DependencyInjector(
            self.artifact_store
        )
        
        # Target generation and validation
        self.target_manager = TargetManager()
        
        self.results = []
        self.completed_modules = set()  # (dataset, module_id) tuples
        self.start_time = None
        self.end_time = None
    
    def run(self) -> bool:
        """
        Run the full research campaign.
        
        Returns:
            bool: True if successful
        """
        self.start_time = time.time()
        
        # Load datasets
        self.logger.info("Loading datasets...")
        datasets = self.dataset_registry.load_all()
        
        if not datasets:
            self.logger.error("No datasets loaded!")
            return False
        
        self.logger.info(f"Loaded {len(datasets)} datasets: {list(datasets.keys())}")
        
        # Discover modules
        self.logger.info("Discovering modules...")
        modules = self.module_loader.discover_modules()
        
        # Build execution order with proper dependency resolution
        if self.modules_filter:
            self.logger.info(f"Filtering to modules: {self.modules_filter}")
            execution_order = []
            
            for module_id in self.modules_filter:
                module_order = get_execution_order(modules, root_module=module_id)
                execution_order.extend(module_order)
            
            # Remove duplicates while preserving order
            seen = set()
            execution_order = [
                m for m in execution_order
                if not (m["module_id"] in seen or seen.add(m["module_id"]))
            ]
        else:
            execution_order = get_execution_order(modules)
        
        self.logger.info(f"Found {len(modules)} modules")
        self.logger.info(f"Execution order: {[m['module_id'] for m in execution_order]}")
        
        if not execution_order:
            self.logger.error("No modules to execute!")
            return False
        
        # Start campaign
        self.logger.start_campaign(
            "Research Campaign v1",
            execution_order,
            list(datasets.keys())
        )
        
        # Run each dataset first (recommended: execute by dataset)
        all_stats = []
        index_data = []
        
        # Module 112 is a cross-timeframe module.
        # It must execute once using all available timeframes together.
        MULTI_TIMEFRAME_MODULES = {"112"}
        
        for dataset_name, market_df in datasets.items():
            self.logger.info(f"\n{'='*60}")
            self.logger.info(f"PROCESSING DATASET: {dataset_name}")
            self.logger.info(f"{'='*60}")
            
            for module_info in execution_order:
                module_id = module_info['module_id']
                
                # Module 112 is executed separately after all timeframe-specific
                # modules have completed, because it requires 15m + 1h + 4h together.
                if module_id in MULTI_TIMEFRAME_MODULES:
                    continue
                
                module_name = module_info.get('module_name', module_id)
                dependencies = module_info.get('depends_on', [])
                
                self.logger.start_module(module_id, dataset_name)
                
                try:
                    start_time = time.time()
                    
                    # Validate dependencies
                    try:
                        self.dependency_injector.validate_dependencies(
                            dataset_name, module_id, dependencies, self.completed_modules
                        )
                    except RuntimeError as e:
                        self.logger.error(f"    Dependency validation failed: {e}")
                        self.results.append({
                            'module_id': module_id,
                            'dataset': dataset_name,
                            'success': False,
                            'duration': 0,
                            'error': str(e)
                        })
                        continue
                    
                    # Create output directory
                    module_output_dir = (
                        self.export_manager.raw_dir / 
                        f"module{module_id}" / 
                        dataset_name
                    )
                    module_output_dir.mkdir(parents=True, exist_ok=True)
                    
                    # Instantiate module
                    module = self.module_loader.instantiate_module(module_info)
                    
                    # ----------------------------------------------------
                    # Inject dependency features / artifacts
                    # ----------------------------------------------------
                    if dependencies:
                        self.dependency_injector.log_dependency_summary(
                            dataset_name, module_id, dependencies
                        )

                        if module_id == "117":
                            # Module 117 requires the original artifact dictionaries
                            # from Modules 109, 114, 115, and 116.
                            dependency_artifacts = {}

                            for dep in dependencies:
                                artifacts = self.artifact_store.get(
                                    dataset_name,
                                    dep
                                )

                                if not artifacts:
                                    raise RuntimeError(
                                        f"Module 117 dependency {dep} has no artifacts "
                                        f"for dataset {dataset_name}"
                                    )

                                dependency_artifacts[dep] = artifacts

                            module.set_dependency_artifacts(dependency_artifacts)

                            self.logger.info(
                                f"    Module 117 adapter received artifacts from "
                                f"{len(dependency_artifacts)} dependencies"
                            )

                        elif module_id == "118":
                            # Module 118 consumes:
                            #   - Module 117 "kept_features" to know WHICH features survived
                            #   - Module 109 "features"     to obtain the ACTUAL feature values
                            dependency_artifacts = {}

                            for dep in dependencies:
                                artifacts = self.artifact_store.get(
                                    dataset_name,
                                    dep
                                )

                                if not artifacts:
                                    raise RuntimeError(
                                        f"Module 118 dependency {dep} has no artifacts "
                                        f"for dataset {dataset_name}"
                                    )

                                dependency_artifacts[dep] = artifacts

                            # ----------------------------------------------------
                            # 117 tells us WHICH features survived
                            # ----------------------------------------------------
                            selected_features = []

                            artifacts_117 = dependency_artifacts.get("117", {})

                            kept_features = artifacts_117.get(
                                "kept_features",
                                []
                            )

                            if kept_features is not None:
                                if isinstance(kept_features, pd.DataFrame):
                                    if "feature" in kept_features.columns:
                                        selected_features = (
                                            kept_features["feature"]
                                            .dropna()
                                            .astype(str)
                                            .tolist()
                                        )
                                elif isinstance(kept_features, (list, tuple)):
                                    selected_features = [
                                        str(f) for f in kept_features
                                    ]

                            # NOTE: Zero KEEP features is a valid research conclusion.
                            # Module 118 must still run to record the transformation
                            # stage conclusion. Do NOT raise here.
                            if not selected_features:
                                self.logger.warning(
                                    f"    Module 118 received zero KEEP features "
                                    f"from Module 117 for dataset {dataset_name}. "
                                    f"Proceeding with empty feature set."
                                )

                            # ----------------------------------------------------
                            # 109 supplies the ACTUAL feature columns
                            # (only if we have features to select)
                            # ----------------------------------------------------
                            if selected_features:
                                artifacts_109 = dependency_artifacts.get("109", {})

                                feature_df = artifacts_109.get("features")

                                if feature_df is None:
                                    raise RuntimeError(
                                        f"Module 118: Module 109 has no 'features' artifact "
                                        f"for dataset {dataset_name}"
                                    )

                                if not isinstance(feature_df, pd.DataFrame):
                                    raise RuntimeError(
                                        f"Module 118: Module 109 'features' is not "
                                        f"a DataFrame for dataset {dataset_name}"
                                    )

                                missing = [
                                    f for f in selected_features
                                    if f not in feature_df.columns
                                ]

                                if missing:
                                    raise RuntimeError(
                                        f"Module 118: selected features are missing from "
                                        f"Module 109 feature data: {missing}"
                                    )

                                selected_feature_df = feature_df[
                                    selected_features
                                ].copy()

                                # Pass the actual feature values into Module 118
                                module.set_dependency_features(
                                    selected_feature_df
                                )

                                self.logger.info(
                                    f"    Module 118 adapter received "
                                    f"{len(selected_features)} KEEP features "
                                    f"from Module 117 (values from Module 109)"
                                )
                            else:
                                # Empty feature selection — pass empty DataFrame
                                module.set_dependency_features(pd.DataFrame())

                                self.logger.info(
                                    f"    Module 118 adapter received "
                                    f"0 KEEP features from Module 117"
                                )

                            # Pass the KEEP list (possibly empty) to the engine
                            module.engine.selected_features = selected_features

                        elif module_id == "119":
                            # Module 119 analyzes the quality of the canonical engineered
                            # features produced upstream by Modules 100, 102, 035 and 037.
                            #
                            # Do NOT consume Module 103's "features" artifact here.
                            # Module 103 exports timestamp + state, whereas Module 119
                            # needs the actual engineered feature values.

                            dependency_artifacts = {}

                            for dep in dependencies:
                                artifacts = self.artifact_store.get(
                                    dataset_name,
                                    dep
                                )

                                if not artifacts:
                                    raise RuntimeError(
                                        f"Module 119 dependency {dep} has no artifacts "
                                        f"for dataset {dataset_name}"
                                    )

                                dependency_artifacts[dep] = artifacts

                            # ----------------------------------------------------
                            # Required canonical feature sources
                            # ----------------------------------------------------
                            required_features = {
                                "100": "trend_score",
                                "102": "compression_score",
                                "035": "hurst_exponent",
                                "037": "normalized_entropy",
                            }

                            feature_parts = []

                            for dep, feature_name in required_features.items():

                                artifacts = dependency_artifacts.get(dep)

                                if not artifacts:
                                    raise RuntimeError(
                                        f"Module 119 requires Module {dep} artifacts "
                                        f"for dataset {dataset_name}"
                                    )

                                feature_df = artifacts.get("features")

                                if feature_df is None:
                                    raise RuntimeError(
                                        f"Module {dep} did not provide a 'features' artifact "
                                        f"for Module 119, dataset {dataset_name}"
                                    )

                                if not isinstance(feature_df, pd.DataFrame):
                                    raise RuntimeError(
                                        f"Module {dep} 'features' artifact is not a DataFrame "
                                        f"for dataset {dataset_name}"
                                    )

                                if feature_name not in feature_df.columns:
                                    raise RuntimeError(
                                        f"Module {dep} 'features' artifact does not contain "
                                        f"required feature '{feature_name}' "
                                        f"for dataset {dataset_name}. "
                                        f"Available columns: {list(feature_df.columns)}"
                                    )

                                feature_parts.append(
                                    feature_df[[feature_name]].reset_index(drop=True)
                                )

                            # ----------------------------------------------------
                            # Construct the actual Module 119 feature matrix
                            # ----------------------------------------------------
                            feature_df = pd.concat(
                                feature_parts,
                                axis=1
                            )

                            # Verify row alignment with the current market dataset.
                            if len(feature_df) != len(market_df):
                                raise RuntimeError(
                                    f"Module 119 feature row count mismatch for "
                                    f"dataset {dataset_name}: "
                                    f"features={len(feature_df)}, "
                                    f"market={len(market_df)}"
                                )

                            # Final structural verification.
                            expected_columns = [
                                "trend_score",
                                "compression_score",
                                "hurst_exponent",
                                "normalized_entropy",
                            ]

                            missing = [
                                col for col in expected_columns
                                if col not in feature_df.columns
                            ]

                            if missing:
                                raise RuntimeError(
                                    f"Module 119 feature construction failed for "
                                    f"dataset {dataset_name}. Missing: {missing}"
                                )

                            self.logger.info(
                                f"    Module 119 adapter constructed "
                                f"{len(feature_df.columns)} canonical features "
                                f"from Modules 100, 102, 035 and 037"
                            )

                            self.logger.info(
                                f"    Module 119 feature columns: "
                                f"{list(feature_df.columns)}"
                            )

                            module.set_dependency_features(feature_df)

                        elif module_id == "120":
                            # Module 120 analyzes temporal stability of the
                            # canonical engineered features.
                            #
                            # It must receive the actual feature values from:
                            #   100 -> trend_score
                            #   102 -> compression_score
                            #   035 -> hurst_exponent
                            #   037 -> normalized_entropy
                            #
                            # Do NOT consume Module 103's "features" artifact.
                            # Module 103 exports state information, not the
                            # canonical feature-value matrix required here.

                            dependency_artifacts = {}

                            for dep in dependencies:
                                artifacts = self.artifact_store.get(
                                    dataset_name,
                                    dep
                                )

                                if not artifacts:
                                    raise RuntimeError(
                                        f"Module 120 dependency {dep} has no artifacts "
                                        f"for dataset {dataset_name}"
                                    )

                                dependency_artifacts[dep] = artifacts

                            # ----------------------------------------------------
                            # Required canonical feature sources
                            # ----------------------------------------------------
                            required_features = {
                                "100": "trend_score",
                                "102": "compression_score",
                                "035": "hurst_exponent",
                                "037": "normalized_entropy",
                            }

                            feature_parts = []

                            for dep, feature_name in required_features.items():

                                artifacts = dependency_artifacts.get(dep)

                                if not artifacts:
                                    raise RuntimeError(
                                        f"Module 120 requires Module {dep} artifacts "
                                        f"for dataset {dataset_name}"
                                    )

                                feature_df = artifacts.get("features")

                                if feature_df is None:
                                    raise RuntimeError(
                                        f"Module {dep} did not provide a 'features' artifact "
                                        f"for Module 120, dataset {dataset_name}"
                                    )

                                if not isinstance(feature_df, pd.DataFrame):
                                    raise RuntimeError(
                                        f"Module {dep} 'features' artifact is not a DataFrame "
                                        f"for Module 120, dataset {dataset_name}"
                                    )

                                if feature_name not in feature_df.columns:
                                    raise RuntimeError(
                                        f"Module {dep} 'features' artifact does not contain "
                                        f"required feature '{feature_name}' "
                                        f"for Module 120, dataset {dataset_name}. "
                                        f"Available columns: {list(feature_df.columns)}"
                                    )

                                feature_parts.append(
                                    feature_df[[feature_name]].reset_index(drop=True)
                                )

                            # ----------------------------------------------------
                            # Construct Module 120 feature matrix
                            # ----------------------------------------------------
                            feature_df = pd.concat(
                                feature_parts,
                                axis=1
                            )

                            # ----------------------------------------------------
                            # Verify row alignment
                            # ----------------------------------------------------
                            if len(feature_df) != len(market_df):
                                raise RuntimeError(
                                    f"Module 120 feature row count mismatch for "
                                    f"dataset {dataset_name}: "
                                    f"features={len(feature_df)}, "
                                    f"market={len(market_df)}"
                                )

                            # ----------------------------------------------------
                            # Final structural verification
                            # ----------------------------------------------------
                            expected_columns = [
                                "trend_score",
                                "compression_score",
                                "hurst_exponent",
                                "normalized_entropy",
                            ]

                            missing = [
                                col
                                for col in expected_columns
                                if col not in feature_df.columns
                            ]

                            if missing:
                                raise RuntimeError(
                                    f"Module 120 feature construction failed for "
                                    f"dataset {dataset_name}. Missing: {missing}"
                                )

                            self.logger.info(
                                f"    Module 120 adapter constructed "
                                f"{len(feature_df.columns)} canonical features "
                                f"from Modules 100, 102, 035 and 037"
                            )

                            self.logger.info(
                                f"    Module 120 feature columns: "
                                f"{list(feature_df.columns)}"
                            )

                            # Pass actual feature values into Module 120
                            module.set_dependency_features(feature_df)

                        elif module_id == "121":
                            # Module 121 analyzes redundancy between the canonical
                            # engineered features produced upstream by:
                            #   100 -> trend_score
                            #   102 -> compression_score
                            #   035 -> hurst_exponent
                            #   037 -> normalized_entropy

                            dependency_artifacts = {}

                            for dep in dependencies:
                                artifacts = self.artifact_store.get(dataset_name, dep)

                                if not artifacts:
                                    raise RuntimeError(
                                        f"Module 121 requires Module {dep} artifacts "
                                        f"for dataset {dataset_name}"
                                    )

                                dependency_artifacts[dep] = artifacts

                            required_features = {
                                "100": "trend_score",
                                "102": "compression_score",
                                "035": "hurst_exponent",
                                "037": "normalized_entropy",
                            }

                            feature_parts = []

                            for dep, feature_name in required_features.items():
                                artifacts = dependency_artifacts.get(dep)

                                feature_df = artifacts.get("features") if artifacts else None

                                if not isinstance(feature_df, pd.DataFrame):
                                    raise RuntimeError(
                                        f"Module 121 requires Module {dep} feature DataFrame "
                                        f"for dataset {dataset_name}"
                                    )

                                if feature_name not in feature_df.columns:
                                    raise RuntimeError(
                                        f"Module 121 requires feature '{feature_name}' "
                                        f"from Module {dep} for dataset {dataset_name}"
                                    )

                                feature_parts.append(
                                    feature_df[[feature_name]].reset_index(drop=True)
                                )

                            feature_df = pd.concat(feature_parts, axis=1)

                            if len(feature_df) != len(market_df):
                                raise RuntimeError(
                                    f"Module 121 feature row count {len(feature_df)} "
                                    f"does not match market rows {len(market_df)} "
                                    f"for dataset {dataset_name}"
                                )

                            expected_columns = [
                                "trend_score",
                                "compression_score",
                                "hurst_exponent",
                                "normalized_entropy",
                            ]

                            if list(feature_df.columns) != expected_columns:
                                raise RuntimeError(
                                    f"Module 121 feature columns mismatch. "
                                    f"Expected {expected_columns}, got {list(feature_df.columns)}"
                                )

                            self.logger.info(
                                "    Module 121 adapter constructed "
                                "4 canonical features from Modules 100, 102, 035 and 037"
                            )
                            self.logger.info(
                                f"    Module 121 feature columns: {list(feature_df.columns)}"
                            )

                            module.set_dependency_features(feature_df)

                        elif module_id == "124":
                            # Module 124 analyzes redundancy among the four canonical
                            # engineered features produced by Modules 100, 102, 035 and 037.

                            dependency_artifacts = {}

                            for dep in dependencies:
                                artifacts = self.artifact_store.get(
                                    dataset_name,
                                    dep
                                )

                                if not artifacts:
                                    raise RuntimeError(
                                        f"Module 124 dependency {dep} has no artifacts "
                                        f"for dataset {dataset_name}"
                                    )

                                dependency_artifacts[dep] = artifacts

                            required_features = {
                                "100": "trend_score",
                                "102": "compression_score",
                                "035": "hurst_exponent",
                                "037": "normalized_entropy",
                            }

                            feature_parts = []

                            for dep, feature_name in required_features.items():
                                artifacts = dependency_artifacts.get(dep)

                                if not artifacts:
                                    raise RuntimeError(
                                        f"Module 124 requires Module {dep} artifacts "
                                        f"for dataset {dataset_name}"
                                    )

                                feature_df = artifacts.get("features")

                                if not isinstance(feature_df, pd.DataFrame):
                                    raise RuntimeError(
                                        f"Module {dep} 'features' artifact is not a DataFrame "
                                        f"for Module 124"
                                    )

                                if feature_name not in feature_df.columns:
                                    raise RuntimeError(
                                        f"Module {dep} 'features' artifact does not contain "
                                        f"required feature '{feature_name}' for Module 124. "
                                        f"Available columns: {list(feature_df.columns)}"
                                    )

                                feature_parts.append(
                                    feature_df[[feature_name]].reset_index(drop=True)
                                )

                            feature_df = pd.concat(feature_parts, axis=1)

                            expected_columns = [
                                "trend_score",
                                "compression_score",
                                "hurst_exponent",
                                "normalized_entropy",
                            ]

                            if list(feature_df.columns) != expected_columns:
                                raise RuntimeError(
                                    f"Module 124 feature construction failed. "
                                    f"Expected {expected_columns}, got {list(feature_df.columns)}"
                                )

                            if len(feature_df) != len(market_df):
                                raise RuntimeError(
                                    f"Module 124 feature row mismatch: "
                                    f"market={len(market_df)}, features={len(feature_df)}"
                                )

                            module.set_dependency_features(feature_df)

                            self.logger.info(
                                "    Module 124 adapter constructed 4 canonical features "
                                "from Modules 100, 102, 035 and 037"
                            )

                        elif module_id == "126":
                            # Module 126 is a feature repository.
                            # It requires each feature DataFrame to contain a timestamp column.

                            feature_dict = {}

                            for dep in dependencies:
                                artifacts = self.artifact_store.get(
                                    dataset_name,
                                    dep
                                )

                                if not artifacts:
                                    raise RuntimeError(
                                        f"Module 126 dependency {dep} has no artifacts "
                                        f"for dataset {dataset_name}"
                                    )

                                feature_df = artifacts.get("features")

                                if not isinstance(feature_df, pd.DataFrame):
                                    raise RuntimeError(
                                        f"Module 126 dependency {dep} does not provide "
                                        f"a valid 'features' DataFrame for dataset {dataset_name}"
                                    )

                                feature_df = feature_df.copy().reset_index(drop=True)

                                # Module 126's repository engine requires timestamp
                                # in every feature DataFrame.
                                if "timestamp" not in feature_df.columns:
                                    if len(feature_df) != len(market_df):
                                        raise RuntimeError(
                                            f"Module 126 dependency {dep} row count "
                                            f"{len(feature_df)} does not match market rows "
                                            f"{len(market_df)} for dataset {dataset_name}"
                                        )

                                    feature_df.insert(
                                        0,
                                        "timestamp",
                                        market_df["timestamp"].reset_index(drop=True)
                                    )

                                else:
                                    feature_df["timestamp"] = (
                                        feature_df["timestamp"]
                                        .reset_index(drop=True)
                                    )

                                # Final row-count validation
                                if len(feature_df) != len(market_df):
                                    raise RuntimeError(
                                        f"Module 126 dependency {dep} row count "
                                        f"{len(feature_df)} does not match market rows "
                                        f"{len(market_df)} for dataset {dataset_name}"
                                    )

                                feature_dict[dep] = feature_df

                                self.logger.info(
                                    f"    Module 126 received {len(feature_df.columns)} "
                                    f"columns from Module {dep} ({len(feature_df)} rows)"
                                )

                            self.logger.info(
                                f"    Module 126 constructed feature_dict from "
                                f"{len(feature_dict)} upstream modules"
                            )

                            self.logger.info(
                                f"    Module 126 feature sources: {list(feature_dict.keys())}"
                            )

                            # IMPORTANT:
                            # Module 126 must be executed through its dedicated
                            # run_repository() interface.
                            result = module.run_repository(
                                feature_dict=feature_dict,
                                dataset_id=f"{module_id}_{dataset_name}",
                                output_dir=module_output_dir,
                            )

                            duration = time.time() - start_time

                            # Store standardized artifacts
                            if hasattr(result, "artifacts") and result.artifacts:
                                self.artifact_store.put(
                                    dataset_name,
                                    module_id,
                                    result.artifacts
                                )

                                self.logger.info(
                                    f"    Stored {len(result.artifacts)} artifacts "
                                    f"for {dataset_name}"
                                )

                            # Export outputs
                            self.export_manager.export_module_outputs(
                                module_id,
                                dataset_name,
                                result,
                                module_output_dir
                            )

                            # Extract statistics
                            stats_df = self.export_manager.consolidate_statistics(
                                module_id,
                                dataset_name,
                                result
                            )

                            # Add metadata

                                    "module_version",
                                    "unknown"
                                ),
                                "timestamp": datetime.now().isoformat(),
                                "dependencies": ",".join(dependencies),
                                "has_features": self.artifact_store.has(
                                    dataset_name,
                                    module_id,
                                    "features"
                                ),
                            })

                            self.logger.module_result(
                                module_id,
                                dataset_name,
                                success,
                                duration,
                                len(stats_df)
                            )

                            # ----------------------------------------------------
                            # Build and export Module 203 report
                            # ----------------------------------------------------

                            report = self.summary_builder.build_module_report(
                                module_id,
                                module_info,
                                result,
                                dataset_name,
                                duration
                            )

                            self.export_manager.export_report(
                                f"{module_id}_{dataset_name}",
                                report
                            )

                            self.results.append({
                                "module_id": module_id,
                                "dataset": dataset_name,
                                "success": success,
                                "duration": duration,
                                "stats_count": len(stats_df)
                            })

                            # Only successful modules satisfy downstream
                            # dependencies.
                            if success:
                                self.completed_modules.add(
                                    (dataset_name, module_id)
                                )

                            self.logger.info(
                                f"    Module 203 completed for {dataset_name}"
                            )

                            # IMPORTANT:
                            # Prevent Module 203 from falling through to the
                            # generic run_dataframe() path.
                            continue

                        elif module_id == "204":
                            # ----------------------------------------------------
                            # Module 204 is a file-driven feature redundancy/synergy
                            # module.
                            #
                            # It consumes Module 203's predictive feature matrix.
                            # Do NOT use the generic run_dataframe() path.
                            # ----------------------------------------------------

                            module203_output_dir = (
                                self.export_manager.raw_dir
                                / "module203"
                                / dataset_name
                            )

                            matrix_path = (
                                module203_output_dir
                                / "predictive_feature_matrix.parquet"
                            )

                            # ----------------------------------------------------
                            # Validate Module 203 input
                            # ----------------------------------------------------

                            if not matrix_path.exists():
                                raise FileNotFoundError(
                                    f"Module 204 requires Module 203 predictive "
                                    f"feature matrix for {dataset_name}:\n"
                                    f"{matrix_path}"
                                )

                            predictive_matrix_df = pd.read_parquet(matrix_path)

                            if predictive_matrix_df.empty:
                                raise ValueError(
                                    f"Module 204 predictive matrix is empty for "
                                    f"{dataset_name}:\n{matrix_path}"
                                )

                            if "timestamp" not in predictive_matrix_df.columns:
                                raise ValueError(
                                    f"Module 204 predictive matrix missing "
                                    f"'timestamp' column for {dataset_name}:\n"
                                    f"{matrix_path}"
                                )

                            # Feature columns = everything except timestamp/OHLCV.
                            ignored_columns = {
                                "timestamp",
                                "date",
                                "time",
                                "index",
                                "close",
                                "open",
                                "high",
                                "low",
                                "volume",
                            }

                            feature_columns = [
                                c for c in predictive_matrix_df.columns
                                if c not in ignored_columns
                            ]

                            if not feature_columns:
                                raise ValueError(
                                    f"Module 204 received no usable feature columns "
                                    f"from Module 203 for {dataset_name}.\n"
                                    f"Available columns: "
                                    f"{list(predictive_matrix_df.columns)}"
                                )

                            # ----------------------------------------------------
                            # Validate timestamp alignment against market dataset
                            # ----------------------------------------------------

                            if len(predictive_matrix_df) != len(market_df):
                                raise ValueError(
                                    f"Module 204 row-count mismatch for {dataset_name}: "
                                    f"predictive_matrix={len(predictive_matrix_df)}, "
                                    f"market_data={len(market_df)}"
                                )

                            predictive_timestamp = pd.to_datetime(
                                predictive_matrix_df["timestamp"]
                                .reset_index(drop=True)
                            )

                            market_timestamp = pd.to_datetime(
                                market_df["timestamp"]
                                .reset_index(drop=True)
                            )

                            if not predictive_timestamp.equals(market_timestamp):
                                raise ValueError(
                                    f"Module 204 timestamp alignment failed for "
                                    f"{dataset_name}. Module 203 predictive matrix "
                                    f"and raw market data are not aligned."
                                )

                            self.logger.info(
                                f"    Module 204 received predictive matrix from "
                                f"Module 203: {len(predictive_matrix_df):,} rows, "
                                f"{len(feature_columns)} features"
                            )

                            self.logger.info(
                                f"    Module 204 matrix path: {matrix_path}"
                            )

                            self.logger.info(
                                f"    Module 204 feature columns: {feature_columns}"
                            )

                            # ----------------------------------------------------
                            # IMPORTANT:
                            # Use Module 204's dedicated production interface.
                            # ----------------------------------------------------

                            result = module.run_redundancy_synergy(
                                matrix_path=matrix_path,
                                dataset_id=f"{module_id}_{dataset_name}",
                                output_dir=module_output_dir
                            )

                            duration = time.time() - start_time

                            # ----------------------------------------------------
                            # Standardized artifact bookkeeping
                            # ----------------------------------------------------

                            if hasattr(result, "artifacts") and result.artifacts:
                                self.artifact_store.put(
                                    dataset_name,
                                    module_id,
                                    result.artifacts
                                )

                                self.logger.info(
                                    f"    Stored {len(result.artifacts)} artifacts "
                                    f"for {dataset_name}"
                                )

                            self.export_manager.export_module_outputs(
                                module_id,
                                dataset_name,
                                result,
                                module_output_dir
                            )

                            stats_df = self.export_manager.consolidate_statistics(
                                module_id,
                                dataset_name,
                                result
                            )

                            stats_df["timestamp"] = datetime.now().isoformat()
                            stats_df["duration"] = duration
                            stats_df["version"] = module_info.get(
                                "module_version",
                                "unknown"
                            )
                            stats_df["has_features"] = self.artifact_store.has(
                                dataset_name,
                                module_id,
                                "features"
                            )
                            stats_df["dataset"] = dataset_name

                            all_stats.append(stats_df)

                            success = result_succeeded(result)

                            index_data.append({
                                "module_id": module_id,
                                "module_name": module_name,
                                "dataset": dataset_name,
                                "status": "PASS" if success else "FAIL",
                                "csv_path": str(module_output_dir),
                                "execution_time": duration,
                                "rows": len(stats_df),
                                "version": module_info.get(
                                    "module_version",
                                    "unknown"
                                ),
                                "timestamp": datetime.now().isoformat(),
                                "dependencies": ",".join(dependencies),
                                "has_features": self.artifact_store.has(
                                    dataset_name,
                                    module_id,
                                    "features"
                                ),
                            })

                            if success:
                                self.completed_modules.add(
                                    (dataset_name, module_id)
                                )

                            self.logger.info(
                                f"    Module 204 completed for {dataset_name}"
                            )

                            # Do not fall through to generic run_dataframe().
                            continue

                        else:
                            feature_df = self.dependency_injector.build_feature_dataframe(
                                dataset_name,
                                dependencies
                            )

                            if feature_df is not None:
                                module.set_dependency_features(feature_df)

                                self.logger.info(
                                    f"    Injected {len(feature_df.columns)} "
                                    f"features from {len(dependencies)} modules"
                                )
                            else:
                                self.logger.warning(
                                    f"    No features found for dependencies: {dependencies}"
                                )
                    
                    # ----------------------------------------------------
                    # Prepare module-specific market DataFrame
                    # ----------------------------------------------------
                    module_market_df = market_df.copy()
                    
                    # ----------------------------------------------------
                    # Ensure module-required research targets
                    # ----------------------------------------------------
                    required_targets = self.target_manager.get_target_requirements(
                        module_info
                    )
                    
                    if required_targets:
                        self.logger.info(
                            f"    Target requirements for Module {module_id}: "
                            f"{required_targets}"
                        )
                        
                        module_market_df = self.target_manager.ensure_targets(
                            module_market_df,
                            required_targets,
                            dataset_id=f"{module_id}_{dataset_name}",
                        )
                        
                        # Verify target generation succeeded
                        missing_targets = [
                            target
                            for target in required_targets
                            if target not in module_market_df.columns
                        ]
                        
                        if missing_targets:
                            raise RuntimeError(
                                f"Module {module_id} target preparation failed. "
                                f"Missing targets: {missing_targets}"
                            )
                        
                        self.logger.info(
                            f"    Verified required targets for Module {module_id}: "
                            f"{required_targets}"
                        )
                    
                    # ----------------------------------------------------
                    # Diagnostic input inspection
                    # ----------------------------------------------------
                    self.logger.info(
                        f"    Module {module_id} input shape: "
                        f"{module_market_df.shape}"
                    )
                    
                    self.logger.info(
                        f"    Module {module_id} input columns: "
                        f"{list(module_market_df.columns)}"
                    )
                    
                    # ----------------------------------------------------
                    # Run module
                    # ----------------------------------------------------
                    result = self.module_loader.run_module(
                        module,
                        module_market_df,
                        dataset_id=f"{module_id}_{dataset_name}",
                        output_dir=module_output_dir
                    )
                    
                    duration = time.time() - start_time
                    
                    # Store artifacts (standardized format) with dataset awareness
                    if hasattr(result, 'artifacts') and result.artifacts:
                        self.artifact_store.put(dataset_name, module_id, result.artifacts)
                        self.logger.info(f"    Stored {len(result.artifacts)} artifacts for {dataset_name}")
                    
                    # Export outputs
                    self.export_manager.export_module_outputs(
                        module_id,
                        dataset_name,
                        result,
                        module_output_dir
                    )
                    
                    # Extract statistics
                    stats_df = self.export_manager.consolidate_statistics(
                        module_id,
                        dataset_name,
                        result
                    )
                    
                    # Add metadata
                    stats_df['timestamp'] = datetime.now().isoformat()
                    stats_df['duration'] = duration
                    stats_df['version'] = module_info.get('module_version', 'unknown')
                    stats_df['has_features'] = self.artifact_store.has(dataset_name, module_id, "features")
                    stats_df['dataset'] = dataset_name
                    
                    all_stats.append(stats_df)
                    
                    # FIXED: Use canonical result_succeeded()
                    success = result_succeeded(result)
                    
                    # Build index entry
                    index_data.append({
                        'module_id': module_id,
                        'module_name': module_name,
                        'dataset': dataset_name,
                        'status': 'PASS' if success else 'FAIL',
                        'csv_path': str(module_output_dir),
                        'execution_time': duration,
                        'rows': len(stats_df),
                        'version': module_info.get('module_version', 'unknown'),
                        'timestamp': datetime.now().isoformat(),
                        'dependencies': ','.join(dependencies),
                        'has_features': self.artifact_store.has(dataset_name, module_id, "features")
                    })
                    
                    self.logger.module_result(
                        module_id,
                        dataset_name,
                        success,
                        duration,
                        len(stats_df)
                    )
                    
                    # Build and export report
                    report = self.summary_builder.build_module_report(
                        module_id,
                        module_info,
                        result,
                        dataset_name,
                        duration
                    )
                    self.export_manager.export_report(
                        f"{module_id}_{dataset_name}",
                        report
                    )
                    
                    self.results.append({
                        'module_id': module_id,
                        'dataset': dataset_name,
                        'success': success,
                        'duration': duration,
                        'stats_count': len(stats_df)
                    })
                    
                    # FIXED: Mark as completed only when execution actually succeeded.
                    # Failed modules must never satisfy downstream dependencies.
                    if success:
                        self.completed_modules.add((dataset_name, module_id))
                    
                except Exception as e:
                    duration = time.time() - start_time
                    self.logger.error(f"    ✗ FAILED: {str(e)}")
                    self.logger.error(traceback.format_exc())
                    
                    self.results.append({
                        'module_id': module_id,
                        'dataset': dataset_name,
                        'success': False,
                        'duration': duration,
                        'error': str(e)
                    })
                    
                    index_data.append({
                        'module_id': module_id,
                        'module_name': module_name,
                        'dataset': dataset_name,
                        'status': 'FAIL',
                        'csv_path': '',
                        'execution_time': duration,
                        'rows': 0,
                        'version': module_info.get('module_version', 'unknown'),
                        'timestamp': datetime.now().isoformat(),
                        'dependencies': ','.join(dependencies),
                        'error': str(e)
                    })
        
        # ============================================================
        # MODULE 112 - CROSS-TIMEFRAME EXECUTION
        # ============================================================
        #
        # Module 112 must receive dependency-enriched feature
        # DataFrames for ALL available timeframes simultaneously.
        #
        # Normal modules run:
        #     dataset -> module
        #
        # Module 112 runs:
        #     all datasets -> module 112
        #
        # ============================================================

        module112_info = None

        for module_info in execution_order:
            if module_info["module_id"] == "112":
                module112_info = module_info
                break

        if module112_info is not None:

            module_id = "112"
            module_name = module112_info.get("module_name", module_id)
            dependencies = module112_info.get("depends_on", [])

            self.logger.info("")
            self.logger.info("=" * 60)
            self.logger.info("PROCESSING MODULE 112 - CROSS-TIMEFRAME")
            self.logger.info("=" * 60)
            self.logger.info(
                f"Module 112 dependencies: {dependencies}"
            )

            try:
                start_time = time.time()

                # ----------------------------------------------------
                # Determine target requirements for Module 112
                # ----------------------------------------------------
                required_targets = self.target_manager.get_target_requirements(
                    module112_info
                )

                if required_targets:
                    self.logger.info(
                        f"  Module 112 target requirements: "
                        f"{required_targets}"
                    )

                # ----------------------------------------------------
                # Validate dependencies for EVERY timeframe
                # and build target-enriched datasets
                # ----------------------------------------------------
                module112_datasets = {}

                for dataset_name, market_df in datasets.items():

                    self.logger.info(
                        f"  Building Module 112 features for {dataset_name}"
                    )

                    self.dependency_injector.validate_dependencies(
                        dataset_name,
                        module_id,
                        dependencies,
                        self.completed_modules
                    )

                    # Start from ORIGINAL market/OHLCV data
                    module_market_df = market_df.copy()

                    # Build dependency features
                    feature_df = self.dependency_injector.build_feature_dataframe(
                        dataset_name,
                        dependencies
                    )

                    if feature_df is None:
                        raise RuntimeError(
                            f"No dependency features found for Module 112 "
                            f"on timeframe {dataset_name}"
                        )

                    # Attach dependency features to the market dataframe
                    module_market_df = pd.concat(
                        [
                            module_market_df.reset_index(drop=True),
                            feature_df.reset_index(drop=True)
                        ],
                        axis=1
                    )

                    # Generate targets using the OHLCV-containing dataframe
                    if required_targets:
                        module_market_df = self.target_manager.ensure_targets(
                            module_market_df,
                            required_targets,
                            dataset_id=f"{module_id}_{dataset_name}",
                        )

                        missing_targets = [
                            target
                            for target in required_targets
                            if target not in module_market_df.columns
                        ]

                        if missing_targets:
                            raise RuntimeError(
                                f"Module 112 target preparation failed "
                                f"on timeframe {dataset_name}. "
                                f"Missing targets: {missing_targets}"
                            )

                        self.logger.info(
                            f"    {dataset_name}: targets ensured "
                            f"{required_targets}"
                        )

                    module112_datasets[dataset_name] = module_market_df

                    self.logger.info(
                        f"    {dataset_name}: "
                        f"{len(module_market_df)} rows, "
                        f"{len(module_market_df.columns)} columns"
                    )

                # ----------------------------------------------------
                # Instantiate Module 112
                # ----------------------------------------------------
                module112 = self.module_loader.instantiate_module(
                    module112_info
                )

                # ----------------------------------------------------
                # Create dedicated output directory
                # ----------------------------------------------------
                module112_output_dir = (
                    self.export_manager.raw_dir
                    / "module112"
                    / "multi"
                )

                module112_output_dir.mkdir(
                    parents=True,
                    exist_ok=True
                )

                # ----------------------------------------------------
                # Run Module 112 ONCE across all timeframes
                # ----------------------------------------------------
                result = module112.run_datasets(
                    module112_datasets,
                    dataset_id="112_multi",
                    output_dir=module112_output_dir
                )

                duration = time.time() - start_time

                # ----------------------------------------------------
                # Store artifacts under a dedicated multi-timeframe key
                # ----------------------------------------------------
                if hasattr(result, "artifacts") and result.artifacts:
                    self.artifact_store.put(
                        "multi",
                        module_id,
                        result.artifacts
                    )

                    self.logger.info(
                        f"    Stored {len(result.artifacts)} artifacts "
                        f"for Module 112"
                    )

                # ----------------------------------------------------
                # Export outputs
                # ----------------------------------------------------
                self.export_manager.export_module_outputs(
                    module_id,
                    "multi",
                    result,
                    module112_output_dir
                )

                # ----------------------------------------------------
                # Extract statistics
                # ----------------------------------------------------
                stats_df = self.export_manager.consolidate_statistics(
                    module_id,
                    "multi",
                    result
                )

                stats_df["timestamp"] = datetime.now().isoformat()
                stats_df["duration"] = duration
                stats_df["version"] = module112_info.get(
                    "module_version",
                    "unknown"
                )
                stats_df["dataset"] = "multi"
                stats_df["has_features"] = (
                    self.artifact_store.has(
                        "multi",
                        module_id,
                        "features"
                    )
                )

                all_stats.append(stats_df)

                # ----------------------------------------------------
                # Determine success
                # ----------------------------------------------------
                success = result_succeeded(result)

                index_data.append({
                    "module_id": module_id,
                    "module_name": module_name,
                    "dataset": "multi",
                    "status": "PASS" if success else "FAIL",
                    "csv_path": str(module112_output_dir),
                    "execution_time": duration,
                    "rows": len(stats_df),
                    "version": module112_info.get(
                        "module_version",
                        "unknown"
                    ),
                    "timestamp": datetime.now().isoformat(),
                    "dependencies": ",".join(dependencies),
                    "has_features": self.artifact_store.has(
                        "multi",
                        module_id,
                        "features"
                    )
                })

                self.logger.module_result(
                    module_id,
                    "multi",
                    success,
                    duration,
                    len(stats_df)
                )

                # ----------------------------------------------------
                # Build and export Module 112 report
                # ----------------------------------------------------
                report = self.summary_builder.build_module_report(
                    module_id,
                    module112_info,
                    result,
                    "multi",
                    duration
                )

                self.export_manager.export_report(
                    "112_multi",
                    report
                )

                self.results.append({
                    "module_id": module_id,
                    "dataset": "multi",
                    "success": success,
                    "duration": duration,
                    "stats_count": len(stats_df)
                })

                # Module 112 completed successfully
                if success:
                    self.completed_modules.add(
                        ("multi", module_id)
                    )

            except Exception as e:

                duration = time.time() - start_time

                self.logger.error(
                    f"    ✗ Module 112 FAILED: {str(e)}"
                )
                self.logger.error(
                    traceback.format_exc()
                )

                self.results.append({
                    "module_id": "112",
                    "dataset": "multi",
                    "success": False,
                    "duration": duration,
                    "error": str(e)
                })

                index_data.append({
                    "module_id": "112",
                    "module_name": module112_info.get(
                        "module_name",
                        "112"
                    ),
                    "dataset": "multi",
                    "status": "FAIL",
                    "csv_path": "",
                    "execution_time": duration,
                    "rows": 0,
                    "version": module112_info.get(
                        "module_version",
                        "unknown"
                    ),
                    "timestamp": datetime.now().isoformat(),
                    "dependencies": ",".join(
                        module112_info.get(
                            "depends_on",
                            []
                        )
                    ),
                    "error": str(e)
                })
        
        self.end_time = time.time()
        total_time = self.end_time - self.start_time
        
        # Save target generation log
        target_log = self.target_manager.get_target_log()
        if target_log:
            target_log_path = self.output_dir / "target_log.json"
            with open(target_log_path, 'w', encoding='utf-8') as f:
                json.dump(target_log, f, indent=2, default=str)
            self.logger.info(f"Saved target log: {target_log_path}")

        if all_stats:
            master_df = pd.concat(all_stats, ignore_index=True)
            self.export_manager.save_master_database(master_df)
            self.logger.info(f"Saved master database: {len(master_df)} rows")
        
        # Save research index
        index_df = pd.DataFrame(index_data)
        self.export_manager.save_research_index(index_df)
        self.logger.info(f"Saved research index: {len(index_df)} entries")
        
        # Save injection log
        injection_log = self.dependency_injector.get_injection_log()
        injection_path = self.output_dir / "injection_log.json"
        with open(injection_path, 'w', encoding='utf-8') as f:
            json.dump(injection_log, f, indent=2, default=str)
        self.logger.info(f"Saved injection log: {injection_path}")
        
        # Generate campaign summary
        summary = self.summary_builder.build_campaign_summary(
            self.results,
            total_time
        )
        summary_path = self.output_dir / "campaign_summary.txt"
        with open(summary_path, 'w', encoding='utf-8') as f:
            f.write(summary)
        self.logger.info(f"Saved campaign summary: {summary_path}")
        
        # Save artifact store summary
        artifact_summary = self.artifact_store.summary()
        artifact_path = self.output_dir / "artifact_summary.json"
        with open(artifact_path, 'w', encoding='utf-8') as f:
            json.dump(artifact_summary, f, indent=2)
        self.logger.info(f"Saved artifact summary: {artifact_path}")
        
        # Finish campaign
        total_success = sum(1 for r in self.results if r.get('success', False))
        self.logger.finish_campaign(len(self.results), total_success, total_time)
        
        return total_success == len(self.results)
