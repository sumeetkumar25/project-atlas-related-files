# atlas/framework/framework_check.py
"""
Framework Health Check
Validates entire Atlas infrastructure before research begins.
"""

import sys
from pathlib import Path
from datetime import datetime
from typing import List, Tuple

# Add project root
sys.path.insert(0, str(Path(__file__).parent.parent.parent))

from atlas.core.console import setup_console

# Configure console encoding before any output
setup_console()

from atlas.framework.registry import AtlasRegistry
from atlas.core.research_result import ResearchResult
from atlas.core.base_research_module import BaseResearchModule

# FIXED: Import from research.export_manager, not core.export_manager
from atlas.research.export_manager import ExportManager
from atlas.runner import ModuleRunner


class FrameworkChecker:
    """
    Comprehensive health check for Project Atlas framework.
    """
    
    def __init__(self):
        self.registry = AtlasRegistry()
        self.checks_passed = 0
        self.checks_failed = 0
        self.checks_warning = 0
        self.issues: List[str] = []
    
    def run_all_checks(self) -> bool:
        """Run complete framework health check."""
        self._print_header()
        
        checks = [
            ("Framework Version", self._check_framework_version, True),
            ("Configuration", self._check_configuration, True),
            ("Registry Modules", self._check_registry_modules, True),
            ("Registry Datasets", self._check_registry_datasets, True),
            ("ResearchResult", self._check_research_result, True),
            ("BaseResearchModule", self._check_base_module, True),
            ("ExportManager", self._check_export_manager, True),
            ("Runner", self._check_runner, True),
            ("Data Files", self._check_data_files, True),
            ("Dependencies", self._check_dependencies, True),
            ("Dataset Consistency", self._check_dataset_consistency, False),
        ]
        
        for check_name, check_func, is_critical in checks:
            passed, message = check_func()
            self._report_check(check_name, passed, message, is_critical)
        
        self._print_summary()
        return self.checks_failed == 0
    
    def _print_header(self):
        print("\n" + "=" * 60)
        print("PROJECT ATLAS")
        print("FRAMEWORK HEALTH CHECK")
        print("=" * 60)
        print(f"\nTimestamp: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        print(f"Python: {sys.version.split()[0]}")
        print()
    
    def _report_check(self, name: str, passed: bool, message: str, is_critical: bool = True):
        if passed:
            icon = "OK"
            status = "PASS"
            self.checks_passed += 1
        elif is_critical:
            icon = "FAIL"
            status = "FAIL"
            self.checks_failed += 1
            self.issues.append(f"{name}: {message}")
        else:
            icon = "WARN"
            status = "WARN"
            self.checks_warning += 1
            self.issues.append(f"{name} (warning): {message}")
        
        print(f"[{icon}] {name:<25} {status}")
        if message and not passed:
            print(f"     -> {message}")
    
    def _check_framework_version(self) -> Tuple[bool, str]:
        try:
            version = self.registry.framework_version
            if version:
                return True, f"v{version}"
            return False, "Version not set"
        except Exception as e:
            return False, str(e)
    
    def _check_configuration(self) -> Tuple[bool, str]:
        try:
            import config
            required_attrs = ['PROJECT_ROOT', 'DATA_DIR', 'RAW_DATA_DIR', 'OUTPUTS_DIR']
            for attr in required_attrs:
                if not hasattr(config, attr):
                    return False, f"Missing config attribute: {attr}"
            return True, "Configuration loaded"
        except ImportError:
            return False, "config.py not found"
        except Exception as e:
            return False, str(e)
    
    def _check_registry_modules(self) -> Tuple[bool, str]:
        try:
            stats = self.registry.get_statistics()
            module_count = stats['total_modules']
            if module_count > 0:
                return True, f"{module_count} modules registered"
            return False, "No modules registered"
        except Exception as e:
            return False, str(e)
    
    def _check_registry_datasets(self) -> Tuple[bool, str]:
        try:
            stats = self.registry.get_statistics()
            dataset_count = stats['total_datasets']
            datasets_file = self.registry.datasets_file
            if dataset_count > 0:
                return True, f"{dataset_count} datasets registered"
            elif datasets_file.exists():
                return False, "Datasets file exists but 0 datasets loaded"
            else:
                return True, "No datasets registered yet"
        except Exception as e:
            return False, str(e)
    
    def _check_research_result(self) -> Tuple[bool, str]:
        try:
            result = ResearchResult(
                module_name="test", module_version="1.0",
                dataset_identifier="test.csv",
                timestamp_start="2026-01-01", timestamp_end="2026-12-31",
                execution_timestamp=datetime.now().isoformat()
            )
            result.add_statistic("test_metric", 1.0)
            summary = result.to_summary_dict()
            if 'test_metric' in summary:
                return True, "ResearchResult working"
            return False, "to_summary_dict() failed"
        except Exception as e:
            return False, str(e)
    
    def _check_base_module(self) -> Tuple[bool, str]:
        try:
            required_methods = ['load_data', 'validate', 'compute', 'export', 'summary', 'run']
            for method in required_methods:
                if not hasattr(BaseResearchModule, method):
                    return False, f"Missing method: {method}"
            return True, "BaseResearchModule valid"
        except Exception as e:
            return False, str(e)
    
    def _check_export_manager(self) -> Tuple[bool, str]:
        """Check ExportManager with current API."""
        try:
            import tempfile

            with tempfile.TemporaryDirectory() as tmpdir:
                result = ResearchResult(
                    module_name="test",
                    module_version="1.0",
                    dataset_identifier="test.csv",
                    timestamp_start="2026-01-01",
                    timestamp_end="2026-12-31",
                    execution_timestamp=datetime.now().isoformat()
                )

                result.add_statistic("test", 1.0)

                exporter = ExportManager(tmpdir)

                stats_df = exporter.consolidate_statistics(
                    "TEST",
                    "test",
                    result
                )

                if stats_df.empty:
                    return False, "Statistics consolidation produced no data"

                report_file = exporter.export_report(
                    "TEST",
                    "Test report"
                )

                if not report_file.exists():
                    return False, "Report file was not created"

                return True, "ExportManager working"

        except Exception as e:
            return False, str(e)
    
    def _check_runner(self) -> Tuple[bool, str]:
        try:
            runner = ModuleRunner()
            if hasattr(runner, 'run_on_datasets'):
                return True, "ModuleRunner valid"
            return False, "Missing run_on_datasets method"
        except Exception as e:
            return False, str(e)
    
    def _check_data_files(self) -> Tuple[bool, str]:
        try:
            import config
            datasets = getattr(config, 'DATASETS', {})
            available = []
            missing = []
            for ds_id, filepath in datasets.items():
                if Path(filepath).exists():
                    size_mb = Path(filepath).stat().st_size / (1024 * 1024)
                    available.append(f"{ds_id} ({size_mb:.1f}MB)")
                else:
                    missing.append(ds_id)
            if missing:
                return False, f"Missing: {', '.join(missing)}"
            if available:
                return True, f"All {len(available)} datasets present"
            return False, "No datasets configured or found"
        except Exception as e:
            return False, str(e)
    
    def _check_dependencies(self) -> Tuple[bool, str]:
        try:
            import pandas, numpy, scipy, statsmodels
            versions = (f"pandas {pandas.__version__}, numpy {numpy.__version__}, "
                       f"scipy {scipy.__version__}, statsmodels {statsmodels.__version__}")
            return True, versions
        except ImportError as e:
            return False, f"Missing dependency: {e}"
    
    def _check_dataset_consistency(self) -> Tuple[bool, str]:
        try:
            import config
            configured = set(getattr(config, 'DATASETS', {}).keys())
            registered = set(self.registry.datasets.keys())
            if not configured and not registered:
                return True, "No datasets configured"
            if not configured:
                return True, "No datasets in config"
            only_configured = configured - registered
            if only_configured and not registered:
                return True, f"{len(configured)} datasets configured, awaiting registration"
            return True, f"{len(configured & registered)} datasets consistent"
        except Exception as e:
            return True, f"Consistency check skipped: {e}"
    
    def _print_summary(self):
        print("\n" + "=" * 60)
        print("HEALTH CHECK SUMMARY")
        print("=" * 60)
        total = self.checks_passed + self.checks_failed + self.checks_warning
        print(f"\nTotal Checks: {total}")
        print(f"Passed: {self.checks_passed}")
        print(f"Failed: {self.checks_failed}")
        if self.checks_warning > 0:
            print(f"Warnings: {self.checks_warning}")
        if self.issues:
            print(f"\nIssues Found:")
            for issue in self.issues:
                print(f"  -> {issue}")
        if self.checks_failed == 0:
            print("\nOVERALL STATUS: READY")
        else:
            print(f"\nOVERALL STATUS: NOT READY ({self.checks_failed} critical failure(s))")
        print("=" * 60 + "\n")


def main():
    checker = FrameworkChecker()
    success = checker.run_all_checks()
    sys.exit(0 if success else 1)


if __name__ == "__main__":
    main()
