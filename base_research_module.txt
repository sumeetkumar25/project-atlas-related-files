"""
Base Research Module

Every research module must inherit from this class.

UPDATED v3.12:
- load_dataframe() copies BEFORE modifying to prevent caller DataFrame mutation
- load_data() and load_dataframe() preserve previously injected features
- set_dependency_features() unconditionally drops duplicate timestamp
- set_dependency_features() merges features into self.df for backward compatibility
- Automatic default artifact creation for legacy modules
- Backward compatibility: self.df = self.market_df for legacy modules
- Separate market_df and feature_df (never overwrite raw data)
- Copy on set_dependency_features() to prevent memory sharing
- Copy on load_data() and load_dataframe()
- Standardized artifact requirements (tuple for immutability)
- Type validation for all artifacts including validation
- Single source of truth for artifact validation
- Explicit metadata export (no double export)
- Enhanced summary with market/feature data sources
"""

from abc import ABC, abstractmethod
from typing import Optional, List, Dict, Any, Tuple, Union
import pandas as pd
import time
from datetime import datetime
from pathlib import Path

from atlas.core.research_result import ResearchResult
from atlas.core.logger import AtlasLogger


class PipelineStepResult:
    """Result of a single pipeline step execution."""
    
    def __init__(self, step_name: str):
        self.step_name = step_name
        self.status: str = "completed"
        self.duration: float = 0.0
        self.warnings: List[str] = []
        self.statistics: Dict[str, Any] = {}
        self.artifacts: Dict[str, Any] = {}
    
    def to_dict(self) -> Dict[str, Any]:
        return {
            "step": self.step_name,
            "status": self.status,
            "duration": round(self.duration, 4),
            "warnings": self.warnings,
            **self.statistics
        }


class BaseResearchModule(ABC):
    """
    Abstract base class for all research modules.
    
    Standard workflow:
    1. load_data() / load_dataframe() → market_df (with copy), preserves features in df
    2. set_dependency_features() → feature_df (with copy) → merges into df
    3. validate()
    4. compute() → execute_pipeline()
    5. export()
    6. summary()
    
    Output convention (frozen):
      outputs/module{N}/{dataset}/
    
    Every module must export artifacts with:
    - features: DataFrame
    - statistics: DataFrame
    - report: str
    - validation: bool (or dict with 'passed' key)
    
    Market data and feature data are kept separate:
    - self.market_df: Raw OHLCV data (copy)
    - self.feature_df: Dependency features from other modules (copy)
    
    Backward Compatibility:
    - self.df is merged market_df + feature_df for legacy modules
    - _create_default_artifacts() creates missing required artifacts
    """
    
    # Standardized artifact requirements (immutable tuple)
    REQUIRED_ARTIFACTS: Tuple[str, ...] = (
        "features",
        "statistics",
        "report",
        "validation",
    )
    
    # Validation artifact must be bool or dict with 'passed' key
    VALIDATION_TYPES = (bool, dict)
    
    # Override in subclasses for custom output folder names
    MODULE_OUTPUT: str = ""
    
    def __init__(self, module_name: str, module_version: str = "1.0.0"):
        self.module_name = module_name
        self.module_version = module_version
        self.logger = AtlasLogger(module_name)
        
        # Separate market and feature data (both are copies)
        self.market_df: Optional[pd.DataFrame] = None
        self.feature_df: Optional[pd.DataFrame] = None
        
        # Backward compatibility alias for legacy modules
        # This will be merged market_df + feature_df after injection
        self.df: Optional[pd.DataFrame] = None
        
        self.result: Optional[ResearchResult] = None
        
        self.pipeline_steps: List[Tuple[str, callable]] = []
        self._register_pipeline_steps()
        
        self._step_timings: Dict[str, float] = {}
        self._step_results: List[PipelineStepResult] = []
    
    def _register_pipeline_steps(self):
        """Override to register (name, method) tuples"""
        pass
    
    def _validate_step(self, step_name: str) -> List[str]:
        """Override to add step-specific validation."""
        return []
    
    def set_dependency_features(self, feature_df: pd.DataFrame) -> None:
        """
        Set dependency features from other modules.
        
        This is called by the research runner to inject features
        from dependent modules. Features are stored separately from
        market data to prevent accidental overwriting.
        
        IMPORTANT: 
        - Stores a copy to prevent memory sharing with ArtifactStore.
        - Drops duplicate 'timestamp' column unconditionally, since
          market data always owns the canonical timestamp.
        - Merges feature_df into self.df for backward compatibility.
        
        Args:
            feature_df: Combined feature DataFrame from dependencies
        """
        self.feature_df = feature_df.copy()

        print("[DEBUG BASE] market columns:", self.market_df.columns.tolist() if self.market_df is not None else None)
        print("[DEBUG BASE] incoming feature columns:", feature_df.columns.tolist())
        print("[DEBUG BASE] copied feature columns:", self.feature_df.columns.tolist())
        print("[DEBUG BASE] market has timestamp:", self.market_df is not None and 'timestamp' in self.market_df.columns)
        print("[DEBUG BASE] feature has timestamp:", 'timestamp' in self.feature_df.columns)

        # Market data always owns the canonical timestamp.
        # Dependency artifacts may include timestamp only for alignment/metadata.
        # Drop it unconditionally to prevent duplicate timestamp columns.
        if 'timestamp' in self.feature_df.columns:
            self.feature_df = self.feature_df.drop(columns=['timestamp'])
        
        # Backward compatibility: merge features into self.df
        if self.market_df is not None:
            self.df = pd.concat(
                [
                    self.market_df.reset_index(drop=True),
                    self.feature_df.reset_index(drop=True)
                ],
                axis=1
            )
        else:
            self.df = self.feature_df.copy()
        
        self.logger.info(
            f"Injected {len(self.feature_df.columns)} dependency features (copy) "
            f"→ df now has {len(self.df.columns)} columns"
        )
    
    def _merge_df(self) -> None:
        """
        Merge market_df and feature_df into df for backward compatibility.
        
        Preserves previously injected features when market data is reloaded.
        """
        if self.market_df is not None:
            if self.feature_df is not None:
                self.df = pd.concat(
                    [
                        self.market_df.reset_index(drop=True),
                        self.feature_df.reset_index(drop=True)
                    ],
                    axis=1
                )
            else:
                self.df = self.market_df.copy()
        elif self.feature_df is not None:
            self.df = self.feature_df.copy()
        else:
            self.df = None
    
    def _create_default_artifacts(self):
        """
        Compatibility layer for legacy modules.
        
        Creates the standardized artifacts required by
        BaseResearchModule if they were not explicitly
        produced by the module itself.
        """
        if self.result is None:
            return
        
        artifacts = self.result.artifacts
        
        # -------------------------------
        # FEATURES
        # -------------------------------
        if "features" not in artifacts:
            if self.feature_df is not None:
                artifacts["features"] = self.feature_df.copy()
                self.logger.info("Created 'features' from feature_df")
            elif self.market_df is not None:
                artifacts["features"] = self.market_df.copy()
                self.logger.info("Created 'features' from market_df")
            else:
                artifacts["features"] = pd.DataFrame()
                self.logger.warning("Created empty 'features' (no data available)")
        
        # -------------------------------
        # STATISTICS
        # -------------------------------
        if "statistics" not in artifacts:
            stats = self.result.statistics or {}
            artifacts["statistics"] = pd.DataFrame([stats])
            self.logger.info("Created 'statistics' from result.statistics")
        
        # -------------------------------
        # REPORT
        # -------------------------------
        if "report" not in artifacts:
            artifacts["report"] = self.summary()
            self.logger.info("Created 'report' from summary()")
        
        # -------------------------------
        # VALIDATION
        # -------------------------------
        if "validation" not in artifacts:
            artifacts["validation"] = {
                "passed": True,
                "errors": self.result.errors or [],
                "warnings": self.result.warnings or []
            }
            self.logger.info("Created 'validation' with default values")
    
    def _validate_artifact_types(self, artifacts: Dict[str, Any]) -> List[str]:
        """
        Validate artifact types.
        
        Returns:
            List of validation errors
        """
        errors = []
        
        # Validate features
        if "features" in artifacts:
            if not isinstance(artifacts["features"], pd.DataFrame):
                errors.append(
                    f"features must be DataFrame, got {type(artifacts['features']).__name__}"
                )
        
        # Validate statistics
        if "statistics" in artifacts:
            if not isinstance(artifacts["statistics"], pd.DataFrame):
                errors.append(
                    f"statistics must be DataFrame, got {type(artifacts['statistics']).__name__}"
                )
        
        # Validate report
        if "report" in artifacts:
            if not isinstance(artifacts["report"], str):
                errors.append(
                    f"report must be str, got {type(artifacts['report']).__name__}"
                )
        
        # Validate validation
        if "validation" in artifacts:
            validation = artifacts["validation"]
            if not isinstance(validation, self.VALIDATION_TYPES):
                errors.append(
                    f"validation must be bool or dict, got {type(validation).__name__}"
                )
            elif isinstance(validation, dict) and "passed" not in validation:
                errors.append(
                    "validation dict must contain 'passed' key"
                )
        
        return errors
    
    def _validate_artifacts(self):
        """Validate that required artifacts are present and have correct types."""
        if self.result is None:
            raise RuntimeError("No result object found")
        
        if not hasattr(self.result, 'artifacts'):
            raise RuntimeError("Result has no artifacts attribute")
        
        artifacts = self.result.artifacts
        
        # Check for missing required artifacts
        missing = []
        for key in self.REQUIRED_ARTIFACTS:
            if key not in artifacts:
                missing.append(key)
        
        if missing:
            raise RuntimeError(
                f"Module {self.module_name} missing required artifacts: {missing}\n"
                f"Available: {list(artifacts.keys())}\n"
                f"Required: {list(self.REQUIRED_ARTIFACTS)}"
            )
        
        # Validate artifact types
        type_errors = self._validate_artifact_types(artifacts)
        if type_errors:
            raise RuntimeError(
                f"Module {self.module_name} has type errors:\n"
                + "\n".join(f"  - {err}" for err in type_errors)
            )
    
    def execute_pipeline(self) -> Dict[str, Any]:
        """Execute all registered pipeline steps in order."""
        all_results = {}
        self._step_results = []
        self._step_timings = {}
        
        for name, step in self.pipeline_steps:
            self.logger.info(f"Step: {name}")
            
            step_start = time.time()
            
            try:
                result = step()
            except Exception as e:
                step_result = PipelineStepResult(name)
                step_result.status = "failed"
                step_result.duration = time.time() - step_start
                step_result.warnings.append(str(e))
                self._step_results.append(step_result)
                self._step_timings[name] = step_result.duration
                raise
            
            step_duration = time.time() - step_start
            self._step_timings[name] = step_duration
            
            step_result = PipelineStepResult(name)
            step_result.duration = step_duration
            
            if isinstance(result, dict):
                all_results.update(result)
                step_result.statistics = result
                # Only update statistics here (not in _execute_workflow)
                if self.result:
                    self.result.statistics.update(result)
            
            warnings = self._validate_step(name)
            step_result.warnings = warnings
            
            for w in warnings:
                self.logger.warning(f"  ⚠ {w}")
            
            self._step_results.append(step_result)
            
            if step_duration > 0.1:
                self.logger.info(f"  Completed in {step_duration:.3f}s")
        
        for name, duration in self._step_timings.items():
            if self.result:
                self.result.add_timing(f"step_{name}", duration)
        
        return all_results
    
    def load_data(self, filepath: str) -> pd.DataFrame:
        """Load and prepare dataset from file"""
        start_time = time.time()
        self.logger.info(f"Loading: {filepath}")
        
        try:
            df = pd.read_csv(filepath, encoding='utf-8')
        except UnicodeDecodeError:
            df = pd.read_csv(filepath)
        
        if 'timestamp' in df.columns:
            df['timestamp'] = pd.to_datetime(df['timestamp'])
            df = df.sort_values('timestamp').reset_index(drop=True)
        
        # Store a copy - framework owns its internal state
        self.market_df = df.copy()
        
        # Preserve previously injected features when merging df
        self._merge_df()
        
        self._init_result(filepath)
        
        self.result.add_timing('load_data', time.time() - start_time)
        self.logger.success(f"Loaded {len(df):,} rows (copy)")
        
        return df
    
    def load_dataframe(self, df: pd.DataFrame, dataset_id: str = "in_memory") -> pd.DataFrame:
        """
        Load data directly from a DataFrame.
        
        IMPORTANT: Copies BEFORE modifying anything.
        The caller's DataFrame is never mutated.
        """
        start_time = time.time()
        self.logger.info(f"Loading from DataFrame: {dataset_id}")
        
        # Copy BEFORE modifying anything
        df = df.copy()
        
        if 'timestamp' in df.columns:
            df['timestamp'] = pd.to_datetime(df['timestamp'])
            df = df.sort_values('timestamp').reset_index(drop=True)
        
        # Framework owns its internal copy
        self.market_df = df.copy()
        
        # Preserve previously injected dependency features
        self._merge_df()
        
        self._init_result(dataset_id)
        
        self.result.add_timing('load_data', time.time() - start_time)
        self.logger.success(f"Loaded {len(df):,} rows from DataFrame (copy)")
        
        return self.market_df
    
    def _init_result(self, dataset_identifier: str):
        """Initialize ResearchResult with dataset fingerprint"""
        self.result = ResearchResult(
            module_name=self.module_name,
            module_version=self.module_version,
            dataset_identifier=dataset_identifier,
            timestamp_start=str(self.market_df['timestamp'].min()) if 'timestamp' in self.market_df.columns else "N/A",
            timestamp_end=str(self.market_df['timestamp'].max()) if 'timestamp' in self.market_df.columns else "N/A",
            execution_timestamp=datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        )
        
        if self.market_df is not None:
            self.result.add_diagnostic('dataset_rows', len(self.market_df))
            self.result.add_diagnostic('dataset_columns', len(self.market_df.columns))
            if 'timestamp' in self.market_df.columns:
                self.result.add_diagnostic('dataset_start', str(self.market_df['timestamp'].min()))
                self.result.add_diagnostic('dataset_end', str(self.market_df['timestamp'].max()))
    
    def validate(self) -> bool:
        """Validate input data quality"""
        start_time = time.time()
        self.logger.section("Data Validation")
        
        valid = True
        required_cols = ['open', 'high', 'low', 'close', 'volume']
        
        for col in required_cols:
            if col not in self.market_df.columns:
                self.result.add_warning(f"Missing column: {col}")
                valid = False
        
        if valid:
            nan_count = self.market_df[required_cols].isna().sum().sum()
            if nan_count > 0:
                self.result.add_warning(f"Found {nan_count} NaN values")
                self.result.add_diagnostic('nan_count', int(nan_count))
        
        self.result.add_timing('validate', time.time() - start_time)
        
        if valid:
            self.logger.success("Validation passed")
        else:
            self.logger.warning("Validation completed with warnings")
        
        return valid
    
    @abstractmethod
    def compute(self) -> Dict[str, Any]:
        """Execute research computations"""
        pass
    
    def get_config_snapshot(self) -> Dict[str, Any]:
        return {
            "module_name": self.module_name,
            "module_version": self.module_version,
        }
    
    def get_protocol_snapshot(self) -> Dict[str, Any]:
        return {}
    
    def export(self, output_dir: str) -> None:
        """Export research results to files"""
        start_time = time.time()
        self.logger.section("Exporting Results")
        
        from atlas.core.export_manager import ExportManager
        exporter = ExportManager(output_dir)
        
        # Export standardized artifacts (skip metadata, handle separately)
        if self.result and hasattr(self.result, 'artifacts'):
            artifacts = self.result.artifacts
            
            # Export all artifacts except metadata (handled separately)
            for name, artifact in artifacts.items():
                if name == "metadata":
                    continue
                if isinstance(artifact, pd.DataFrame):
                    exporter.export_dataframe(artifact, f"{name}.csv")
                elif isinstance(artifact, str):
                    exporter.export_text(artifact, f"{name}.txt")
                elif isinstance(artifact, dict):
                    exporter.export_json(artifact, f"{name}.json")
            
            # Explicitly export metadata if present (single source)
            if "metadata" in artifacts:
                exporter.export_json(artifacts["metadata"], "metadata.json")
        
        exporter.export_statistics(self.result)
        exporter.export_report(self.result, 
                              config_snapshot=self.get_config_snapshot(),
                              protocol_snapshot=self.get_protocol_snapshot(),
                              step_timings=self._step_timings)
        
        self.result.add_timing('export', time.time() - start_time)
        self.logger.success(f"Exported to {output_dir}")
    
    def summary(self) -> str:
        """Generate text summary"""
        lines = []
        lines.append("=" * 60)
        lines.append(f"MODULE: {self.module_name}")
        lines.append(f"VERSION: {self.module_version}")
        lines.append("=" * 60)
        lines.append(f"\nDataset: {self.result.dataset_identifier}")
        lines.append(f"Status: {self.result.status}")
        
        if self._step_timings:
            lines.append(f"\nStep Timing:")
            total = 0
            for name, duration in self._step_timings.items():
                lines.append(f"  {name:<20} {duration:.3f}s")
                total += duration
            lines.append(f"  {'─' * 30}")
            lines.append(f"  {'Total pipeline':<20} {total:.3f}s")
        
        if self.result.statistics:
            lines.append(f"\nStatistics:")
            for key, value in self.result.statistics.items():
                if isinstance(value, float):
                    lines.append(f"  {key}: {value:.6f}")
                else:
                    lines.append(f"  {key}: {value}")
        
        # Show artifacts
        if self.result and hasattr(self.result, 'artifacts'):
            lines.append(f"\nArtifacts:")
            for key in self.REQUIRED_ARTIFACTS:
                if key in self.result.artifacts:
                    artifact = self.result.artifacts[key]
                    if isinstance(artifact, pd.DataFrame):
                        lines.append(f"  ✓ {key}: DataFrame ({len(artifact)} rows, {len(artifact.columns)} cols)")
                    elif isinstance(artifact, str):
                        lines.append(f"  ✓ {key}: Text ({len(artifact)} chars)")
                    elif isinstance(artifact, dict):
                        lines.append(f"  ✓ {key}: Dict ({len(artifact)} keys)")
                    else:
                        lines.append(f"  ✓ {key}: {type(artifact).__name__}")
                else:
                    lines.append(f"  ✗ {key}: MISSING")
        
        # Show data sources
        lines.append(f"\nData Sources:")
        if self.market_df is not None:
            lines.append(f"  market_df: {len(self.market_df)} rows, {len(self.market_df.columns)} cols (copy)")
        else:
            lines.append(f"  market_df: None")
        
        if self.feature_df is not None:
            lines.append(f"  feature_df: {len(self.feature_df)} rows, {len(self.feature_df.columns)} cols (copy)")
        else:
            lines.append(f"  feature_df: None")
        
        # Show backward compatibility alias
        if self.df is not None:
            lines.append(f"  df (merged): {len(self.df)} rows, {len(self.df.columns)} cols")
        
        return "\n".join(lines)
    
    def run(self, filepath: str, output_dir: Optional[str] = None) -> ResearchResult:
        """Execute workflow from file"""
        self.logger.header(self.module_name, "Research Execution")
        self.load_data(filepath)
        return self._execute_workflow(output_dir)
    
    def run_dataframe(self, df: pd.DataFrame,
                      dataset_id: str = "in_memory",
                      output_dir: Optional[str] = None) -> ResearchResult:
        """Execute workflow from DataFrame"""
        self.logger.header(self.module_name, f"Research Execution [{dataset_id}]")
        self.load_dataframe(df, dataset_id)
        return self._execute_workflow(output_dir)
    
    def _execute_workflow(self, output_dir: Optional[str] = None) -> ResearchResult:
        """Common workflow execution with automatic default output directory."""
        if self.market_df is None:
            raise RuntimeError("No market data loaded. Call load_data() or load_dataframe() first.")
        
        if not self.validate():
            self.logger.warning("Proceeding despite validation warnings")
        
        self.logger.section("Running Computation Pipeline")
        pipeline_start = time.time()
        
        try:
            results = self.compute()
            # Note: statistics are already updated in execute_pipeline
            # Only update if compute returns additional results not in pipeline
            if results and self.result:
                for key, value in results.items():
                    if key not in self.result.statistics:
                        self.result.statistics[key] = value
        except Exception as e:
            self.result.status = "failed"
            self.result.errors.append(str(e))
            self.logger.error(f"Computation failed: {e}")
            raise
        
        self.result.add_timing('compute_total', time.time() - pipeline_start)
        
        # ----------------------------------------------------
        # Legacy module compatibility
        # Automatically create standard artifacts
        # ----------------------------------------------------
        self._create_default_artifacts()
        
        # Validate required artifacts
        self._validate_artifacts()
        
        # Default output directory: outputs/module{N}/{dataset}/
        if output_dir is None:
            dataset = Path(self.result.dataset_identifier).stem
            
            if self.MODULE_OUTPUT:
                folder = self.MODULE_OUTPUT
            else:
                folder = self.module_name.replace(" ", "_")
            
            output_dir = str(Path("outputs") / folder / dataset)
        
        self.export(str(output_dir))
        
        self.logger.section("Summary")
        print(self.summary())
        
        self.logger.success("Research complete")
        
        return self.result
