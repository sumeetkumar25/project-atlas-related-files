    @staticmethod
    def _resolve_validation(result) -> bool:
        """
        Resolve the validation status for a module result.

        Resolution order:
            1. result.validation (top-level attribute)
            2. result.artifacts["validation"]
            3. result.statistics["validation_passed"]
            4. result.statistics["software_validation_passed"]
            5. Default: True

        If the resolved value is a dict (the standard validation
        artifact shape emitted by BaseResearchModule), its "passed"
        field is used, falling back to "software_validation_passed",
        then True.

        Returns:
            True if validation is considered passed, False otherwise.

        IMPORTANT:
            Python treats any non-empty dict as truthy. Without the
            explicit isinstance-based unwrap below, a validation
            artifact of the form {"passed": False, ...} would evaluate
            as True and the module would be reported as PASSED
            despite failing validation. That is exactly the failure
            mode this helper exists to prevent.
        """
        validation = getattr(result, "validation", None)

        if validation is None and hasattr(result, "artifacts"):
            validation = result.artifacts.get("validation")

        if validation is None:
            statistics = getattr(result, "statistics", {})
            if not isinstance(statistics, dict):
                statistics = {}
            validation = statistics.get(
                "validation_passed",
                statistics.get("software_validation_passed", True),
            )

        if isinstance(validation, dict):
            validation = validation.get(
                "passed",
                validation.get("software_validation_passed", True),
            )

        return bool(validation)

        if self.timeframes_filter is not None:
            requested = set(self.timeframes_filter)
            available = set(datasets.keys())
            unknown = requested - available
            if unknown:
                raise ValueError(
                    f"Unknown timeframe(s): {sorted(unknown)}. "
                    f"Available: {sorted(available)}"
                )
            datasets = {
                name: df
                for name, df in datasets.items()
                if name in requested
            }
            self.logger.info(
                f"Timeframe filter applied: {list(datasets.keys())}"
            )
        
        if not datasets:
            self.logger.error("No datasets loaded!")
            return False

        self.logger.info("Discovering modules...")

                module_order = get_execution_order(modules, root_module=module_id)
                execution_order.extend(module_order)
            
            # Remove duplicates while preserving order
            seen = set()
            execution_order = [
                m for m in execution_order
                if not (m["module_id"] in seen or seen.add(m["module_id"]))
            ]
        else:
            execution_order = get_execution_order(modules)
        
        self.logger.info(f"Found {len(modules)} modules")
        self.logger.info(f"Execution order: {[m['module_id'] for m in execution_order]}")

            self.logger.error("No modules to execute!")
